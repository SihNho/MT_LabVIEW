class Stop(Exception):
    """A FATAL gate. The stage stops; `close()` still runs the hygiene tail."""


# C6 RESULT LINE (docs/session-protocol.md, user-approved 2026-09-24). `Stage.summary()` prints
# `RESULT {"schema":"result-line/1",...}` as its LAST line, from its own gate counts; bgrun / guard_peer / the
# runner / audit_cycle read ONLY that line (their body scans are gone). The two exit paths summary() cannot see are
# covered here: an UNCAUGHT exception (excepthook -> RESULT FAIL at exit, first_fail = the exception text) and gates
# recorded AFTER the last summary (re-printed at exit). A Stage that never reached summary() and no exception ->
# nothing printed, the exit code decides (a helper-only import must not invent a verdict).
import atexit                                                                      # noqa: E402
import protocol as _protocol                                                       # noqa: E402

_C6 = {"stage": None, "sig": None, "exc": None}


def _c6_line(stage, extra_fail=None):
    n_fail = len(stage.fails) if stage is not None else 0
    n_pass = len(stage.passes) if stage is not None else 0
    first = extra_fail or (getattr(stage, "exc_text", None) if stage is not None else None) \
        or (stage.fails[0] if stage is not None and stage.fails else None)
    if extra_fail:
        n_fail += 1
    arts = []
    if stage is not None:
        sv = stage.R.get("saves") or {}
        if sv.get("md5") and os.path.exists(stage.work):
            arts = [{"path": stage.work, "md5": sv["md5"]}]
    return _protocol.result_line(_protocol.make_result(n_pass, n_fail, first, arts))


def _c6_excepthook(etype, value, tb, _orig=sys.excepthook):
    if not issubclass(etype, (SystemExit, KeyboardInterrupt)):
        _C6["exc"] = "{0}: {1}".format(etype.__name__, str(value))[:200]
    elif issubclass(etype, KeyboardInterrupt):
        _C6["exc"] = "KeyboardInterrupt"
    if _C6["exc"]:                                   # card chat-T3: the structured STOP line of an uncaught exception
        try:
            sys.stdout.write(_protocol.stop_line("traceback", _C6["exc"], exc=etype.__name__) + "\n")
            sys.stdout.flush()
        except Exception:                                                          # noqa: BLE001
            pass
    _orig(etype, value, tb)


def _c6_atexit():
    st = _C6["stage"]
    try:
        if _C6["exc"]:
            line = _c6_line(st, extra_fail=_C6["exc"])
        elif st is not None and _C6["sig"] is not None and _C6["sig"] != (len(st.passes), len(st.fails)):
            line = _c6_line(st)
        else:
            return
        sys.stdout.write(line + "\n")
        sys.stdout.flush()
    except Exception:                                                              # noqa: BLE001
        pass


def _c6_thread_excepthook(args, _orig=getattr(__import__("threading"), "excepthook", None)):
    """card chat-T3: an exception in a THREAD bypasses sys.excepthook (threading.excepthook, Python 3.8+); it is a real
    stop too, so it prints the same structured STOP line (kind traceback, the thread's name) - found by the gate-flip report
    (tools/bench/gate_flips_chat_t3.json SAFETY row diag_c106b_live.log: a reader-thread traceback inside a passing run)."""
    try:
        if not issubclass(args.exc_type, SystemExit):
            msg = "{0}: {1}".format(args.exc_type.__name__, str(args.exc_value))[:200]
            sys.stdout.write(_protocol.stop_line("traceback", msg, exc=args.exc_type.__name__,
                                                 thread=getattr(args.thread, "name", None)) + "\n")
            sys.stdout.flush()
    except Exception:                                                              # noqa: BLE001
        pass
    if _orig is not None:
        _orig(args)


sys.excepthook = _c6_excepthook
if hasattr(__import__("threading"), "excepthook"):
    __import__("threading").excepthook = _c6_thread_excepthook
atexit.register(_c6_atexit)


def md5(path):
    """build_opfsinnertunnelconnect_v0.py:227 verbatim."""
    h = hashlib.md5()
    with open(path, "rb") as f:
        for b in iter(lambda: f.read(1 << 20), b""):
            h.update(b)
    return h.hexdigest()


def version_bytes(path, n=1 << 16):
    """`19 00 80 00` = LV2019, `26 00 80 00` = LV2026 (CLAUDE.md rule 1) - diag_s2_scaffold.py:133-140."""
    import re
    with open(path, "rb") as f:
        head = f.read(n)
    return sorted({m.group(0)[0] for m in re.finditer(rb"(?s).\x00\x80\x00", head)})


def private_bytes():
    """LabVIEW's private bytes - the COMPANION of the handle count, which is blind to VI Server refnums
    (docs/toolkit-capabilities.md:484-485). build_opfsinnertunnelconnect_v0.py:270 verbatim."""
    try:
        r = subprocess.run(["powershell", "-NoProfile", "-Command",
                            "(Get-Process LabVIEW -ErrorAction SilentlyContinue | Select-Object -First 1)"
                            ".PrivateMemorySize64"], capture_output=True, text=True, timeout=60)
        return int((r.stdout or "").strip() or 0)
    except Exception:                                                              # noqa: BLE001
        return None

class Stage(object):
    """One build/diagnostic stage: pin, copy, edit, gate, save, clean up, report."""

    # ------------------------------------------------------------------ construction
    def __init__(self, input_vi, input_md5, name, fresh=True, work_name=None, pins=DEFAULT_PINS,
                 deadline_min=45.0, reserve_s=300.0, out_json=None, preload=True, work_dir=CLAUDEDEV,
                 task=""):
        self.input_vi = input_vi
        self.input_md5 = input_md5
        self.name = name
        self.fresh = fresh
        self.pins = tuple(pins or ())
        self.preload_original = preload
        self.deadline_s = float(deadline_min) * 60.0
        self.reserve_s = float(reserve_s)
        self.t0 = time.time()
        self.stamp = time.strftime("%Y%m%d_%H%M%S")
        self.work_dir = work_dir
        self.work = os.path.join(work_dir, work_name or "{0}_{1}.vi".format(name, self.stamp))
        self.out_json = out_json or os.path.join(BENCH, "{0}.json".format(name))
        self.passes, self.fails, self.facts, self.rows = [], [], [], []
        self.softs = []                   # card chat-S2: LOG-only mismatches (gateclass), recorded, not failures
        self.scratches = []
        self.planned_rows = []            # `plan_rows()`; NOT `self.rows`, which is the non-gate ROW record
        self.sym = {}                     # from_decision symbols: "$name" -> what an earlier row created
        self._rowcheck_done = False
        self._nodes = None
        self._preload = None
        self.started = False
        self.R = {"script": name, "stamp": self.stamp, "input": input_vi, "input_md5_pin": input_md5,
                  "work": self.work, "task": task, "no_vi_was_run": True,
                  "verification_level": "STRUCTURAL unless a gate says otherwise",
                  "es_timeline": [], "ops": [], "saves": {}, "H": {}}

    # ------------------------------------------------------------------ reporting primitives
    def head(self, text):
        print("\n---------- {0}".format(_a(text)), flush=True)

    def gate(self, label, ok, detail="", fatal=False, kind=None):
        ok = bool(ok)
        if not ok:
            # card chat-S2 (PD327, user 2026-10-03): a LOG-only mismatch (tools/gateclass.py, the ONE table) is recorded in
            # tools/bench/gate_soft_log.jsonl and the run CONTINUES - printed `  SOFT  `, never `  FAIL  `, not counted as
            # a fail, never fatal. Every other failing gate is unchanged (FAIL, fatal stays fatal).
            c = _gateclass.classify_gate(label, detail, kind)
            if c["verdict"] == "log":
                self.softs.append(label)
                _gateclass.record_classified(label, c, detail)
                print(_a("  SOFT  {0}{1}  [LOG-only: {2}]".format(label, ("  " + str(detail)) if detail else "",
                                                                   c["rule"])), flush=True)
                return True
        (self.passes if ok else self.fails).append(label)
        print(_a("  {0}  {1}{2}".format("PASS" if ok else "FAIL", label,
                                        ("  " + str(detail)) if detail else "")), flush=True)
        if not ok and fatal:
            raise Stop(label)
        return ok

    def fact(self, line):
        self.facts.append(str(line))
        print(_a("  FACT  {0}".format(line)), flush=True)
        return line

    def row(self, label, observed, expected=None):
        """A NON-GATE record. Deliberately NOT at the start of the line, so a row reporting an expected
        FAIL cannot arm `guard_peer.FAILURE_RE` - the gate belongs to the comparison, not to the row."""
        rec = {"label": label, "observed": observed, "expected": expected}
        self.rows.append(rec)
        print(_a("  ROW   {0} : observed={1} expected={2}".format(label, observed, expected)), flush=True)
        return rec

    # ------------------------------------------------------------------ card 123-7 (PD247(e)): the census helper
    def census_snapshot(self, target=None):
        """{uid: class} of every GObject of `target` (default the work copy) - ONE report_all run. {} in a dry run."""
        if getattr(g.report_all, "_dry", False):
            return {}
        return dict((int(o["uid"]), str(o["class"])) for o in g.report_all(target or self.work, "GObject"))

    def census_gate(self, label, before, after, declared, unpredicted=None):
        """Compare the measured NEW-object census (after - before, by class) with `declared` {class: n} (e.g. the plan's
        `<plan>_pred.json` census block). DRY run -> prints 'UNVERIFIED-DRY' and returns None: a stubbed census is never a
        PASS. Real run -> a gate (PASS iff every class of either side matches), returns the measured {class: n}.
        card 159-a4-1 (PD387(e)): two SimCensus snapshots (sim_census_snapshot, a stagesim state) are compared in a dry run
        too - the gate runs on the SIMULATED end. `unpredicted` = census_unpredicted_reason(pred): a declaration that does not
        cover every op of the plan is UNPREDICTED like an empty one, never a partial PASS."""
        dry = getattr(g.report_all, "_dry", False)
        sim = isinstance(before, SimCensus) and isinstance(after, SimCensus)
        if not declared or unpredicted:
            # card 134-1 (PD287(a)): an EMPTY declared set predicts nothing - CEN2 passed on two empty sets
            # (stage_d1_ring_p3b2a_scratch_c133_6.log:354). Never a PASS: printed UNPREDICTED, returns the measured census.
            got = None
            if (sim or not dry) and isinstance(before, dict) and isinstance(after, dict):
                got = {}
                for u in set(after) - set(before):
                    got[after[u]] = got.get(after[u], 0) + 1
            why = "declared set empty {0!r}".format(declared) if not declared else "declaration does not cover the plan: {0}".format(unpredicted)
            print(_a("  UNPREDICTED  {0}  ({1}; measured {2})".format(
                label, why, json.dumps(got, sort_keys=True) if got is not None else "not in a dry run")), flush=True)
            self.R.setdefault("unpredicted", []).append(label)
            return got
        if (dry and not sim) or not isinstance(before, dict) or not isinstance(after, dict):
            print(_a("  UNVERIFIED-DRY  {0}  (census not measured in a dry run; declared {1})".format(
                label, json.dumps(declared, sort_keys=True))), flush=True)
            return None
        new = set(after) - set(before)
        got = {}
        for u in new:
            got[after[u]] = got.get(after[u], 0) + 1
        diff = dict((c, (got.get(c, 0), int(declared.get(c, 0)))) for c in set(got) | set(declared or {})
                    if got.get(c, 0) != int((declared or {}).get(c, 0)))
        det = {"measured": got, "declared": declared, "diff (measured, declared)": diff}
        if sim:
            det["source"] = "stagesim simulated end (sim_census_snapshot), card 159-a4-1"
        self.gate(label, not diff, det, kind="census")
        return got

    def safe(self, label, fn, default=None):
        try:
            return fn(), ""
        except Exception as e:                                                     # noqa: BLE001
            self._last_exc = e                    # card chat-F2: _op reads its standard outcome (fncontract)
            msg = "{0}: {1}".format(type(e).__name__, str(e)[:250])
            self.fact("{0} raised {1}".format(label, msg))
            return default, msg

    def left_s(self):
        return self.deadline_s - (time.time() - self.t0) - self.reserve_s

    def dump(self):
        self.R["gates"] = {"pass": len(self.passes), "fail": len(self.fails), "failing": self.fails,
                           "soft": list(getattr(self, "softs", []))}
        self.R["facts"] = self.facts
        self.R["rows"] = self.rows
        self.R["elapsed_s"] = round(time.time() - self.t0, 1)
        with open(self.out_json, "w", encoding="utf-8") as f:
            json.dump(self.R, f, indent=1, default=str)

    def summary(self):
        print("\n" + "=" * 100, flush=True)
        print("=== GATES: {0} pass / {1} fail{2}".format(
            len(self.passes), len(self.fails),
            ("; failing: " + ", ".join(self.fails)) if self.fails else ""), flush=True)
        if getattr(self, "softs", None):
            print("=== SOFT (LOG-only, not failures, gate_soft_log.jsonl): {0}: {1}".format(len(self.softs), ", ".join(self.softs)),
                  flush=True)
        print("JSON: {0}   elapsed {1:.1f} s".format(self.out_json, time.time() - self.t0), flush=True)
        _C6["stage"], _C6["sig"] = self, (len(self.passes), len(self.fails))
        print(_c6_line(self), flush=True)                  # C6: the RESULT line is the LAST line
        return 0 if not self.fails else 1

    # ------------------------------------------------------------------ start
    def file_facts(self, tag, path):
        rec = {"path": path, "exists": os.path.exists(path)}
        if rec["exists"]:
            rec["md5"], rec["size"] = md5(path), os.path.getsize(path)
            rec["version_bytes"] = version_bytes(path)
            self.fact("{0}: {1} md5 {2} size {3} B; version bytes {4}".format(
                tag, os.path.basename(path), rec["md5"], rec["size"], rec["version_bytes"]))
        else:
            self.fact("{0}: {1} IS NOT ON DISK".format(tag, os.path.basename(path)))
        return rec

    def pin_check(self, when):
        """Every md5 pin, reported one per line. Returns True iff all hold."""
        allok, seen = True, {}
        for label, path, want in self.pins:
            have = md5(path) if os.path.exists(path) else "MISSING"
            seen[label] = have
            allok = allok and (have == want)
            self.fact("PIN {0:<6} {1:<16} {2}  (want {3}) {4}".format(
                when, label, have, want, "OK" if have == want else "DIFFERS"))
        self.R["pins_" + when.lower()] = seen
        return allok

    def restart(self):
        """Standing restart authority (CLAUDE.md 3). diag_s2_scaffold.fresh:155."""
        bp = mod("bench_prep")
        before, _ = self.safe("handles before restart", bp.labview_handles)
        self.R["H"]["handles_before_restart"] = before
        self.fact("LabVIEW handle count BEFORE the restart: {0!r}".format(before))
        self.safe("restart_labview", bp.restart_labview)
        g.reset()
        time.sleep(3.0)
        after, _ = self.safe("handles after restart", bp.labview_handles)
        self.R["H"]["handles_after_restart"] = after
        self.fact("LabVIEW handle count AFTER the restart: {0!r}".format(after))

    def start(self):
        """FILES ONLY -> restart -> Preload the ORIGINAL read-only -> the dated WORK copy -> open it."""
        self.head("[0] FILES ONLY - the input's md5, the pins, the claudeDev listing")
        # THE LISTING IS TAKEN FIRST, before any fatal gate: a start() that STOPS at a pin used to leave
        # `claudedev_before` unset, and close()'s H6 then read every file in claudeDev as "added".
        self.R["claudedev_before"] = sorted(os.path.basename(p)
                                            for p in glob.glob(os.path.join(CLAUDEDEV, "*.vi")))
        self.fact("claudeDev holds {0} .vi file(s) BEFORE the run".format(len(self.R["claudedev_before"])))
        got = md5(self.input_vi) if os.path.exists(self.input_vi) else "MISSING"
        self.R["input_md5_before"] = got
        self.gate("K1 input md5 == {0}".format(self.input_md5), got == self.input_md5,
                  "got {0}".format(got), fatal=True)
        self.pin_check("BEFORE")
        # The detail carries the HASH **and** the path: a gate that prints only the path cannot tell
        # "absent" from "present with the wrong bytes" (the c87 review's §2, accepted).
        orig = md5(ORIGINAL) if os.path.exists(ORIGINAL) else "MISSING - no such file"
        self.gate("K2 the ORIGINAL's md5 is {0}".format(ORIG_MD5), orig == ORIG_MD5,
                  "got {0} at {1}".format(orig, ORIGINAL), fatal=True)
        if self.fresh:
            self.head("[1] RESTART LabVIEW (standing authority; fresh ~31,500 handles)")
            self.restart()
        if self.preload_original:
            self._preload_enter()
        self.head("[2] THE DATED WORK COPY under claudeDev - the input is never opened for writing")
        shutil.copyfile(self.input_vi, self.work)
        time.sleep(0.4)
        self.gate("K3 the work copy is byte-identical to the input",
                  md5(self.work) == got, os.path.basename(self.work), fatal=True)
        self.safe("ensure_loaded(work)", lambda: g.ensure_loaded(self.work))
        self.es("after open")
        self.started = True
        self.dump()

        """A terminal named by (node uid, terminal name, frame diagram uid) -> the LIVE index triple
        (Diagram idx, Nodes[] idx, Terminals[] idx), every index read off the machine just before use and the
        node's uid ECHOED back. A Selector/Loop tunnel's OUTER terminal is addressed as a terminal of its
        STRUCTURE node, found by GEOMETRY (a structure on the same diagram at the tunnel's left x, or the nearest
        structure up-left of it) - a HEURISTIC, reported in the returned `how`, and made safe by requiring the
        name to resolve to exactly one terminal of the right direction on that structure."""
        t = self.work
        uid, how = int(end["uid"]), "node"
        diags = g.report_all(t, "Diagram")
        didx = next((d["i"] for d in diags if int(d["uid"]) == int(end["diagram"])), None)
        if didx is None:
            raise RuntimeError("diagram #{0} not in report_all('Diagram')".format(end["diagram"]))
        labels = g.node_labels(t, didx)
        uids = [r["uid"] for r in labels]
        if end["owner_class"] in ("SelectorTunnel", "Tunnel", "LoopTunnel") and end["term_class"] == "OuterTerminal":
            pos = dict((o["uid"], (o["pos"], o["class"])) for o in (objs or []))
            tp = pos[uid][0]
            cands = [u for u in uids if pos.get(u, (None, ""))[1] in self.STRUCTS]
            left = [u for u in cands if pos[u][0][0] == tp[0] and pos[u][0][1] <= tp[1]]
            upl = sorted((u for u in cands if pos[u][0][0] <= tp[0] and pos[u][0][1] <= tp[1]),
                         key=lambda u: (-pos[u][0][0], -pos[u][0][1]))
            pick = left or upl[:1]
            if not pick:
                raise RuntimeError("no structure found for tunnel #{0}".format(uid))
            uid, how = pick[0], "structure #{0} of tunnel #{1} (geometry: {2})".format(
                pick[0], end["uid"], "left edge x" if left else "nearest up-left")
        if uid not in uids:
            raise RuntimeError("#{0} not in Diagram[{1}].Nodes[]".format(uid, didx))
        nidx = uids.index(uid)
        echo, rows = g.node_terms_uid(t, didx, nidx)
        if echo != uid:
            raise RuntimeError("uid echo {0!r} != #{1}".format(echo, uid))
        # Pre-decided 173 + 174 (cycle 72 firefighter): ONLY the staged field `verify_term_uid` selects the uid path.
        # Candidate ends from jev_candidates carry `term_uid` at FIRST wiring, when the terminal is still unwired,
        # and r2 (stage_d1_l7_1b_r2.log:41-45) died because that field switched the branch. `term_uid` is ignored
        # here; the second pass / Is Broken? / gates set `verify_term_uid` after the wire has landed.
        if end.get("verify_term_uid"):
            ti, _a = match_term_uid(end["verify_term_uid"], mod("allterms").read_terms(t)[0], rows, is_source)
            return (didx, nidx, ti), "{0}; terminal by uid #{1} (wire {2})".format(how, end["verify_term_uid"], _a["wire_uid"])
        hits = [r for r in rows if r["name"] == end["term"] and bool(r["is_source"]) == bool(is_source)]
        if len(hits) > 1:
            hits = [r for r in hits if not r["wire"]]
        if len(hits) != 1:
            raise RuntimeError("terminal {0!r} ({1}) on #{2}: {3} matches".format(
                end["term"], "source" if is_source else "sink", uid, len(hits)))
        return (didx, nidx, int(hits[0]["i"])), how

    def resolve(self, v):
        """`$name` / `$name.key` -> self.sym; `$wire_of:<uid>:<term i>` -> that terminal's LIVE wire uid (read now,
        never carried). Lists and dicts are resolved recursively; anything else is returned unchanged."""
        if isinstance(v, dict):
            return dict((k, self.resolve(x)) for k, x in v.items())
        if isinstance(v, list):
            return [self.resolve(x) for x in v]
        if not (isinstance(v, str) and v.startswith("$")):
            return v
        if v.startswith("$wire_of:"):
            _w, uid, ti = v.split(":")
            _loc, rows = self.wired_terminals(int(uid), tag="resolve {0}".format(v))
            return int([r for r in rows if int(r["i"]) == int(ti)][0]["wire"])
        name, _dot, key = v[1:].partition(".")
        x = self.sym[name]
        return x[key] if key else x

    def copy_in(self, cls, uid, dest_diagram_uid, position, tag=""):
        """COPY a primitive that already exists on the donor into Diagram #dest by `OpMoveByIndex_v0` (the op of
        `gscript.copy_by_index`, duplicate=True, UID guard) + `move_in` + junk purge. PRECONDITION (the op is
        hard-wired to the NI Moving-Objects pair): `self.work` IS `g.MOVE_DST` and `g.MOVE_SRC` holds the donor's
        bytes - the stage file arranges both. copy_by_index itself is not called because it saves and file-copies
        the Target, which a stage that has more rows to execute must not do. Returns the new node's uid."""
        if os.path.normcase(self.work) != os.path.normcase(g.MOVE_DST):
            raise Stop("copy_in needs work == gscript.MOVE_DST (got {0})".format(self.work))
        order = [int(o["uid"]) for o in g.report_all(g.MOVE_SRC, cls)]
        idx = order.index(int(uid))
        before = set(int(o["uid"]) for o in g.report_all(self.work, cls))

        def _copy():
            lab = json.load(open(os.path.join(BENCH, "opmovebyindex_labels.json"), encoding="utf-8"))
            vi = g.op(g.OP_MOVE_INDEX)
            vi.SetControlValue(lab["class_name"], cls)
            vi.SetControlValue(lab["index"], int(idx))
            vi.SetControlValue(lab["duplicate"], True)
            vi.SetControlValue(lab["traverse_target"], 1)
            g._run(vi)
            return int(vi.GetControlValue(lab["selected_uid"]))
        rec = self._op("copy_in", _copy, "{0}[{1}] #{2}".format(cls, idx, uid))
        if rec["result"] != int(uid):
            raise Stop("copy_in UID guard: selected {0!r} != #{1}".format(rec["result"], uid))
        new = sorted(set(int(o["uid"]) for o in g.report_all(self.work, cls)) - before)
        self.fact("COPY #{0} -> new {1} uid(s) {2!r}".format(uid, cls, new))
        if len(new) != 1:
            raise Stop("copy_in: {0} new {1} object(s), expected 1: {2!r}".format(len(new), cls, new))
        self.junk_purge("{0} after copy".format(tag))
        di = [int(o["uid"]) for o in g.report_all(self.work, "Diagram")].index(int(dest_diagram_uid))
        self.move_in(new[0], di, tuple(position))
        self.junk_purge("{0} after move_in".format(tag), hints=[di, 0])
        return new[0]

    def rule_check(self, d):
        """PRE-DECIDED 143: the op the row names must be the op `jev_pairs.op_rule` names for its classes, unless
        the row carries `op_override` (a reason). Returns (rule_op, rule_variant, why)."""
        import jev_pairs
        ex = d["exec"]

        op, var, why = jev_pairs.op_rule(c)
        if (op, var) != (d.get("op"), d.get("variant")) and not d.get("op_override"):

    def from_decision(self, record, tag="fd"):
        """PLAN STEP 6 / Pre-decided 142-143: execute a decision record (tools/jev_pairs.write_record shape).
        action 'wire'   -> the row's `op` (connect_nested | connect_terminals | wire_sr | connect_from_wire |
                           fs_inner_tunnel_connect) on the `exec` ends {uid, term, term_class, owner_class, diagram};
                           wire_sr also needs exec.sr {loop_uid, loop_class, right_uid, right_uids}. The op is
                           checked against `jev_pairs.op_rule` first (`rule_check`); `es_probe` reads ExecState
                           right after the connect, after 2 s, and after the purge (review c68-p6d §4).
        action 'delete' -> exec.wire_uid is deleted; 'retire' -> exec {class, uid} is deleted.
        action 'copy' -> `copy_in` (exec {cls, uid, dest_diagram, pos}); 'add_shift_reg' -> `add_sr_row`;
        'create_const_on_term' -> `const_row`. A row with `as` stores its result in self.sym for later `$` refs.
        'llm' / 'skip' / anything else -> REPORTED, never executed. Every mutation goes through `_op`, so the node
        census is marked and the measured junk `Invoke` is purged after it. Returns one dict per decision."""
        objs = [dict(o, pos=tuple(o["pos"])) for o in g.report_all(self.work, "GObject")]
        out = []
        for i, d0 in enumerate(record.get("decisions", [])):
            row = {"i": i, "row_key": d0.get("row_key"), "action": d0.get("action"), "op": d0.get("op"),
                   "variant": d0.get("variant"), "id": d0.get("id")}
            act = d0.get("action")
            try:
                d = self.resolve(d0) if act in ("wire", "delete", "retire", "copy", "add_shift_reg",
                                                "create_const_on_term") else d0
                ex = d.get("exec") or {}
                if act == "delete":
                    row["result"] = self.delete_wire(int(ex["wire_uid"]), tag)
                elif act == "retire":
                    row["result"] = self.delete_object(ex["class"], int(ex["uid"]), tag)
                elif act == "copy":
                    row["result"] = self.copy_in(ex["cls"], ex["uid"], ex["dest_diagram"], ex["pos"], d0.get("id"))
                elif act == "add_shift_reg":
                    row["result"] = self.add_sr_row(ex, d0.get("id") or tag)
                elif act == "create_const_on_term":
                    row["result"] = self.const_row(ex, d0.get("id") or tag)
                elif act == "wire":
                    row["op_rule"] = self.rule_check(d)
                    row.update(self._wire_row(d, objs))
                    if d.get("es_probe"):
                        row["es_probe"] = [self.es("{0} right after the connect".format(d0.get("id")))]
                        time.sleep(2.0)
                        row["es_probe"].append(self.es("{0} after 2 s".format(d0.get("id"))))
                    self.junk_purge("{0} row {1}".format(tag, i))
                    if d.get("es_probe"):
                        row["es_probe"].append(self.es("{0} after the purge".format(d0.get("id"))))
                        self.row("P6D-probe {0} ES now/2s/purged".format(d0.get("id")), row["es_probe"],
                                 "recorded")
                else:
                    row["result"] = "NOT EXECUTED (action {0!r})".format(act)
                if d0.get("as") and act in ("copy", "add_shift_reg", "create_const_on_term"):
                    self.sym[d0["as"]] = row["result"]
            except Exception as e:                                                 # noqa: BLE001
                row["error"] = "{0}: {1}".format(type(e).__name__, str(e)[:240])
            res = row.get("result") if isinstance(row.get("result"), dict) else {}
            row["failed_layer"] = "execution" if row.get("error") or res.get("err") \
                else (None if act in ("wire", "delete", "retire", "copy", "add_shift_reg", "create_const_on_term")
                      else (d0.get("evidence") or {}).get("failed_layer"))
            self.row("from_decision row {0} {1}".format(i, row.get("id")), {k: row.get(k) for k in
                     ("action", "op", "variant", "how", "result", "error", "failed_layer")})
            out.append(row)
        self.R["from_decision"] = out
        self.dump()
        return out

    def _wire_row(self, d, objs):
        ex, op = d["exec"], d["op"]
        s, t = ex["src"], ex["dst"]
        if op in ("connect_nested", "connect_terminals"):
            (sd, sn, st), hs = self.address(s, True, objs)
            (dd, dn, dt), hd = self.address(t, False, objs)
            N = mod("build_opconnectnested_v1")
            lab = json.load(open(N.MAP_OUT, encoding="utf-8"))
            if op == "connect_terminals" and sd == dd == 0:
                rec = self._op("connect_terminals", lambda: g.connect_terminals(self.work, dn, dt, sn, st),
                               "N[{0}].t{1} <- N[{2}].t{3}".format(dn, dt, sn, st))
            else:
                rec = self._op("connect_nested_v1", lambda: N.connect_nested_v1(self.work, dd, dn, dt, sd, sn, st, lab),
                               "D[{0}].N[{1}].t{2} <- D[{3}].N[{4}].t{5}".format(dd, dn, dt, sd, sn, st))
            return {"how": [hs, hd], "result": rec}
        if op == "wire_sr":
            sr, variant = ex["sr"], d["variant"]
            body, is_src = (t, False) if variant == "LeftIn" else (s, True)
            reg_end = s if variant == "LeftIn" else t
            if int(body["diagram"]) != int(reg_end["diagram"]):
                raise RuntimeError("body end on diagram {0}, register inside on {1}".format(body["diagram"],
                                                                                            reg_end["diagram"]))
            (bd, bn, bt), hb = self.address(body, is_src, objs)
            cls = sr.get("loop_class", "WhileLoop")
            li = next((r["i"] for r in g.report_all(self.work, cls) if int(r["uid"]) == int(sr["loop_uid"])), None)
            k = list(sr["right_uids"]).index(int(sr["right_uid"]))
            echo = g.shift_reg(self.work, li, k, class_name=cls).get("uid")
            if echo != int(sr["right_uid"]):
                raise RuntimeError("shift_reg[{0}][{1}] echo {2!r} != #{3}".format(li, k, echo, sr["right_uid"]))
            rec = self._op("wire_sr", lambda: g.wire_sr(variant, self.work, li, k, node_index=bn, term_index=bt,
                                                        class_name=cls),
                           "{0} loop[{1}] reg[{2}] node[{3}].t{4}".format(variant, li, k, bn, bt))
            return {"how": [hb, "loop #{0} idx {1}, reg {2} echo #{3}".format(sr["loop_uid"], li, k, echo)],
                    "result": rec}
        if op == "fs_inner_tunnel_connect":
            (sd, sn, st), hs = self.address(s, True, objs)
            return {"how": [hs], "result": self.fs_inner_tunnel_connect(int(t["uid"]), sd, sn, st)}
        if op == "connect_from_wire":
            return self._cfw_row(d, objs)
        raise RuntimeError("op {0!r} has no executor in from_decision".format(op))

    def _cfw_row(self, d, objs):

        how = [hd, "w{0}[{1}] owned by #{2}".format(w, hit[0]["i"], s["uid"])]
        out = {"how": how, "result": rec}
        if variant == "tunnel_outer" and s.get("owner_class") == "LoopTunnel":
            self.junk_purge("cfw before orphan")      # the connect's stray Invoke, before delete re-marks the census

            def _new_tunnel():

                nu = int(new[0]["owner_uid"]) if new else None

                             "es_after": self.es("after orphan delete #{0}".format(uid))}
            self.fact("PD146 orphan #{0}: {1}".format(uid, out["orphan"]))

    def save_route(self, exec_state, broken_ok):
        """WHICH SAVE ROUTE, as a pure function so it is self-testable without LabVIEW.

def fixtures_check(bad_md5s=(), before_listing=None):
    """REVIEW c68-h6 (archive/peer/2026-09-24-c68-h6-fixture-listing.md §1/§4, accepted): after a run that swapped

        a = md5(live) if os.path.exists(live) else "MISSING"
        b = md5(bak) if os.path.exists(bak) else "MISSING"

        ok = ok and good
        print(_a("  {0}  FX {1} == its .ORIG.bak and the .bak is not a run's bytes  live {2} bak {3}".format(

                                                                                     now)), flush=True)

def match_term_uid(term_uid, all_rows, node_rows, is_source):
    """Pre-decided 173 (cycle 71): a terminal addressed by its TERMINAL UID after its first resolution, because a
    terminal's NAME changes on move and on wiring (`stage_d1_l7_1b.log:274`: a new left SR's outer terminal read
    'total data array out' before its wire landed and '' after). Pure function, no LabVIEW.
    `all_rows` = `allterms.read_terms` rows (term_uid, wire_uid, is_source, ...); `node_rows` = `gscript.node_terms`
    rows of the node the connect op addresses (i, name, is_source, wire) - which may be the STRUCTURE carrying the
    terminal (a shift register's outer terminal sits in the loop's Terminals[]). The uid is found in the whole-VI
    table, and its WIRE uid (a unique object) picks the one node row of the same direction. An unwired terminal or
    anything other than exactly one hit RAISES - never a fallback to the name. Returns (terminal index, all-row)."""
    a = [r for r in all_rows if int(r["term_uid"]) == int(term_uid)]
    if len(a) != 1:
        raise RuntimeError("terminal uid #{0}: {1} row(s) in the whole-VI terminal table".format(term_uid, len(a)))
    w = int(a[0]["wire_uid"] or 0)
    if not w or bool(a[0]["is_source"]) != bool(is_source):
        raise RuntimeError("terminal uid #{0}: wire {1}, is_source {2} - uid addressing needs a wired terminal of "
                           "direction {3}".format(term_uid, w, a[0]["is_source"], bool(is_source)))
    hits = [r for r in node_rows if int(r.get("wire") or 0) == w and bool(r["is_source"]) == bool(is_source)]
    if len(hits) != 1:
        raise RuntimeError("terminal uid #{0} (wire {1}): {2} matching row(s) on the addressed node".format(
            term_uid, w, len(hits)))
    return int(hits[0]["i"]), a[0]


def uid_edges(G, kinds=("wire", "fs")):
    """A graph's edges keyed by TERMINAL UID, not by name (review archive/peer/2026-09-24-c68-m4a-p6b-rename.md §4,
    measured tools/bench/q_m4a_diffuid.log: a name-keyed diff reported 13/12 renamed-only edges on unchanged
    uids). `sr` is excluded by default: vigraph pairs registers by equal TOP, and a new register added at an
    occupied TOP makes the pairing ambiguous (same log, U3/U6) - report it, never gate on it."""
    V = mod("vigraph")
    out = set()
    for k, a, b, _i in G["edges"]:
        if k in kinds:
            out.add((k, V.key_parts(a)[0], int((G["rows"].get(a) or {}).get("term_uid") or 0),
                     V.key_parts(b)[0], int((G["rows"].get(b) or {}).get("term_uid") or 0)))
    return out

# (141-2 scratch: base 'output array' terminals 28004/28979 of the deleted #27928/#28916 came back as terminals of the NEW

# SUPERSEDED AS THE GATE KEY by card chat-T6 (docs/d1/result-gates.md): term_identity_gates keys by POSITION (block "POSITION KEYS" below);

def _iv(v):
    try:
        return int(v)
    except (TypeError, ValueError):
        return str(v)


def term_key(r):
    """A terminal row's identity: (term uid, owner uid, term name)."""
    return (_iv(r["term_uid"]), _iv(r.get("owner_uid")), str(r.get("term_name") or ""))


def wire_keys(rows):
    """{(wire uid, frozenset of its SOURCE rows' term_key)} over the wired rows. A recycled wire uid on a new source is a
    different key, so it counts as one lost + one new wire, never as 'kept'."""
    src = {}
    for r in rows:
        w = _iv(r.get("wire_uid") or 0)
        if w in (0, "", "0", "None"):
            continue
        src.setdefault(w, set())
        if r.get("is_source"):
            src[w].add(term_key(r))
    return set((w, frozenset(s)) for w, s in src.items())


def term_identity_gates(base, sim, real, renames=None, bind=None):
    """TD / unwired / D of a stage's end (base = graph rows before, sim = the simulated end rows, real = the read-back end
    rows). Card chat-T6 (docs/d1/result-gates.md): keyed by POSITION - see term_identity_pos, whose dict this returns. `renames` (an old
    pb-licence/1 `renames` list, licence_renames) is still accepted and only REPORTED (`licence_renames_listed`): a name change
    is INFO, licensed or not. `bind` = {"term": {sim uid: real uid}, "obj": {sim uid: real uid}} maps the sim end's created
    objects onto the real ones, so D compares the NEW wires as sets, not only as counts. The name-keyed code that stood here
    (cards 141-3 / 147-b1-2) is kept below as term_identity_names, for reference and comparison only."""
    return term_identity_pos(base, sim, real, bind=bind, renames=renames)


def term_identity_names(base, sim, real):
    """card 141-3 (PD325(b)) as it stood until card chat-T6 - identity (term uid, owner uid, NAME), wires (wire uid, source
    identities). NOT a gate any more (docs/d1/result-gates.md); kept so a test can show what the name key would have said."""
    wv = lambda rows: dict((term_key(r), _iv(r.get("wire_uid") or 0)) for r in rows)   # noqa: E731
    on = lambda w: w not in (0, "", "0", "None")                                       # noqa: E731
    bk, sk, rk = wv(base), wv(sim), wv(real)
    lost, gone = set(bk) - set(rk), set(bk) - set(sk)
    unw = sorted(k for k, w in bk.items() if on(w) and k not in gone and on(sk.get(k, 0)) and not on(rk.get(k, 0)))
    ru = set(k[0] for k in rk)
    bw, sw, rw = wire_keys(base), wire_keys(sim), wire_keys(real)
    new, lost_w, sim_new, sim_lost = rw - bw, bw - rw, sw - bw, bw - sw
    return {"td_ok": not unw and lost == gone, "unwired": unw, "lost": sorted(lost), "gone": sorted(gone),
            "recycled": sorted(k[0] for k in lost if k[0] in ru), "raw_lost": sorted(set(k[0] for k in bk) - ru),
            "d_ok": len(new) == len(sim_new) and len(lost_w) == len(sim_lost),
            "new": sorted(str(k[0]) for k in new), "lost_w": sorted(str(k[0]) for k in lost_w),
            "sim_new": sorted(str(k[0]) for k in sim_new), "sim_lost_w": sorted(str(k[0]) for k in sim_lost)}

# cards 147-b1-2 / 147-c3-2) had to whitelist each batch's renames. S1 (PD337) already ADDRESSES plan terminals by position; the
# RESULT gates now compare by it too:
#   terminal = POSITION KEY (node, index among the node's rows in read order, term class, direction) - addrcheck.pos_table, the
#              index of addrcheck.triple; the name is a report column (`renames`, INFO; a licence's list is only echoed);
#   wire     = (frozenset of its source keys, frozenset of its sink keys) - never its uid; a deleted uid LabVIEW re-issues is INFO.
# Nothing true slips: a base key the sim keeps but the real end lacks = lost (TD FAIL); a kept key that now holds ANOTHER term uid =
# `moved` (TD FAIL - an index / class swap at one node); a key wired in base and in the sim end but not in real = unwired (TD FAIL);
# a wire whose source or sinks change = one lost + one new wire the sim does not have (D FAIL: lost wires and the new wires between
# kept base nodes are compared as SETS, the rest by count - kept_nodes). Pure functions, no LabVIEW.
# Offline test: tools/bench/selftest_t6_poskey.py.
def _on(w):
    return w not in (0, "", "0", "None", None)


def _pos_rows(rows, ob=None, tb=None):
    """{position key: row} of one terminal table (first row of a term uid). The key's node is mapped through `ob` (object
    uids) and `tb` (term uids - a front-panel terminal is its own node): the sim end's created objects onto the real ones."""
    ob, tb = ob or {}, tb or {}
    t = mod("addrcheck").pos_table(rows)
    seen, out = set(), {}
    for r in rows or []:
        u = r["term_uid"]
        if u in seen or u not in t:
            continue
        seen.add(u)
        node, i, c, s = t[u]
        n = _iv(node)
        out[(ob.get(n, tb.get(n, n)), i, c, s)] = r
    return out


def wire_ends_pos(rows, ob=None, tb=None):
    """{wire uid: (frozenset of source position keys, frozenset of sink position keys)} - a wire as its two ends."""
    by = {}
    for k, r in _pos_rows(rows, ob, tb).items():
        w = _iv(r.get("wire_uid") or 0)
        if _on(w):
            by.setdefault(w, (set(), set()))[0 if r.get("is_source") else 1].add(k)
    return dict((w, (frozenset(a), frozenset(b))) for w, (a, b) in by.items())


def kept_nodes(B, gone=()):
    """The BASE nodes that keep at least one base position key (not all in `gone`). Only these nodes' positions are compared
    across the sim end and the real end: a CREATED object's terminal order in the sim end is the simulator's, not LabVIEW's
    (ring P4 s03: the sim lists #29396's 'index' where LabVIEW reads 'new element/subarray', selftest_t6_poskey.log), so its
    wires count by number and its names are left as read."""
    g = set(gone)
    return set(k[0] for k in B if k not in g)


def _bind_maps(bind):
    b = bind or {}
    return (dict((_iv(k), _iv(v)) for k, v in (b.get("term") or {}).items()),
            dict((_iv(k), _iv(v)) for k, v in (b.get("obj") or {}).items()))


def term_identity_pos(base, sim, real, bind=None, renames=None):
    """TD / unwired / D on position keys (block comment above). Returns the keys term_identity_gates always returned -
    {td_ok, unwired, lost, gone, recycled, raw_lost, d_ok, new, lost_w, sim_new, sim_lost_w, d_new, d_lost, reuse_noted,
    reuse_bad, renames, unlicensed_renames, td_key} - plus moved / extra / d_sets / licence_renames_listed:
      gone = base keys absent from the sim end (the plan's deletes); lost = base keys absent from real, plus a GONE key that
      real holds with another term uid (`recycled`: LabVIEW re-used the owner uid); moved = kept keys whose real term uid is
      not the base one; unwired = base keys wired in base and in the sim end, not wired (or absent) in real;
      td_ok = not unwired and not moved and lost == gone;
      d_ok = lost wires == the sim's lost wires (sets, base space), the new wires whose every end is on a kept base node
      (kept_nodes) == the sim's (sets; `d_new_kept_nodes` = their two sizes), and |new| == |sim new| (a wire touching a
      created object counts by number - kept_nodes says why); renames (INFO) = kept keys whose real name differs from the sim
      end's; reuse_noted (INFO) = deleted wire uids present again at the end; extra (INFO) = real keys on a kept base node that
      neither base nor the sim end has."""
    tb, ob = _bind_maps(bind)
    B, S, R = _pos_rows(base), _pos_rows(sim), _pos_rows(real)
    nm = lambda r: str(r.get("term_name") or "")                                      # noqa: E731
    wv = lambda r: _iv(r.get("wire_uid") or 0)                                        # noqa: E731
    gone = set(B) - set(S)
    both = set(B) & set(R)
    moved = sorted(k for k in both - gone if _iv(R[k]["term_uid"]) != _iv(B[k]["term_uid"]))
    recycled = sorted(k for k in both & gone if _iv(R[k]["term_uid"]) != _iv(B[k]["term_uid"]))
    lost = (set(B) - set(R)) | set(recycled)
    unw = sorted(k for k in set(B) - gone if _on(wv(B[k])) and _on(wv(S[k])) and not (k in R and _on(wv(R[k]))))
    ren = sorted((k, nm(S[k]), nm(R[k])) for k in both - gone if k in S and nm(S[k]) != nm(R[k]))
    kn = kept_nodes(B, gone)
    Sm = _pos_rows(sim, ob, tb) if bind is not None else S
    extra = sorted(k for k in set(R) - set(B) - set(Sm) if k[0] in kn)
    be, se, re_ = wire_ends_pos(base), wire_ends_pos(sim, ob, tb) if bind is not None else wire_ends_pos(sim), wire_ends_pos(real)
    bw, sw, rw = set(be.values()), set(se.values()), set(re_.values())
    new, lost_w, sim_new, sim_lost = rw - bw, bw - rw, sw - bw, bw - sw
    on_kept = lambda E: set(p for p in E if all(k[0] in kn for k in p[0] | p[1]))   # noqa: E731
    d_ok = lost_w == sim_lost and on_kept(new) == on_kept(sim_new) and len(new) == len(sim_new)
    uids = lambda E, P: sorted(str(w) for w, p in E.items() if p in P)                # noqa: E731
    deleted = set(be) - set(se)
    noted = [{"wire": w, "base_ends": [sorted(x) for x in be[w]], "real_ends": [sorted(x) for x in re_[w]]}
             for w in sorted(deleted & set(re_), key=str)]
    ru = set(_iv(r["term_uid"]) for r in real or [])
    return {"td_ok": not unw and not moved and lost == gone, "unwired": unw, "lost": sorted(lost), "gone": sorted(gone),
            "moved": moved, "recycled": recycled, "extra": extra,
            "raw_lost": sorted(set(_iv(r["term_uid"]) for r in base or []) - ru),
            "d_ok": d_ok, "d_new_kept_nodes": [len(on_kept(new)), len(on_kept(sim_new))], "new": uids(re_, new), "lost_w": uids(be, lost_w), "sim_new": uids(se, sim_new),
            "sim_lost_w": uids(be, sim_lost), "d_new": len(new), "d_lost": len(lost_w), "reuse_noted": noted, "reuse_bad": [],
            "renames": ren, "unlicensed_renames": [], "licence_renames_listed": sorted(renames or {}),
            "td_key": "position (node, index, term class, direction)"}


def licence_renames(lic):
    """{(term uid, owner uid): (from, to)} from a pb-licence/1 dict's `renames` list ([{node, term_uids, from?, to?}]);
    from/to None = any name. None when the dict has no `renames` key. Since card chat-T6 (docs/d1/result-gates.md) a licence's renames are
    only READ and echoed (term_identity_gates `licence_renames_listed`); no gate needs them."""
    if not isinstance(lic, dict) or "renames" not in lic:
        return None
    out = {}
    for e in lic.get("renames") or []:
        for u in e.get("term_uids") or []:
            out[(_iv(u), _iv(e["node"]))] = (e.get("from"), e.get("to"))
    return out


def pb_rename_rows(base, sim, real, renames=None, bind=None):
    """(rows, restored, unlicensed) for PB's build4 - card chat-T6 (docs/d1/result-gates.md): every real row on a KEPT BASE node (kept_nodes)
    whose POSITION KEY (_pos_rows) the sim end also has gets the sim end's NAME, so PB's node|class|name|ordinal key
    (vigraph.py:244-245) ranks it as the prediction did; wires are untouched, so a wiring difference still shows in the cdiff.
    A created object's rows keep their real names (its sim terminal order is not LabVIEW's - kept_nodes; NG checks those
    names). restored = [(term uid, owner uid, real name, sim name)] (INFO); unlicensed is always [] (renames - an old licence
    list - is accepted and ignored). `bind` is accepted for symmetry with term_identity_gates and not needed here."""
    B, S = _pos_rows(base), _pos_rows(sim)
    kn = kept_nodes(B, set(B) - set(S))
    fix, restored = {}, []
    for k, r in _pos_rows(real).items():
        if k in S and k[0] in kn:
            a, b = str(r.get("term_name") or ""), str(S[k].get("term_name") or "")
            if a != b:
                fix[_iv(r["term_uid"])] = S[k].get("term_name")
                restored.append((_iv(r["term_uid"]), _iv(r.get("owner_uid")), a, b))
    rows = [dict(r, term_name=fix[_iv(r["term_uid"])]) if _iv(r["term_uid"]) in fix else r for r in real]
    return rows, sorted(restored), []

# real-lost term, FAILS. S1's names come from the graph file the D4 JSON pins, after its md5 is checked.
# Pure functions, no LabVIEW. Offline test: tools/bench/selftest_d4_l2b2a.py.
D4_FAIL_KEYS = ("only_sim_terms", "only_sim_edges", "only_real_edges", "dangling_sim_only", "dangling_real_only",
                "unbound")


def d4_load(path):
    """The D4 JSON ({s1: {path, md5}, scope_nodes}) plus `scope` (set), `s1_md5_got` and `names` = {node: {name:
    count}} over S1's terminal rows (deduplicated by term uid). `names` is EMPTY when S1's md5 is not the pinned one:
    the caller gates on `s1_md5_got`."""
    d = json.load(open(path, encoding="utf-8"))
    s1 = os.path.join(ROOT, d["s1"]["path"].replace("/", os.sep))
    got, names, seen = md5(s1), {}, set()
    scope = set(int(n) for n in d["scope_nodes"])
    if got == d["s1"]["md5"]:
        for r in json.load(open(s1, encoding="utf-8"))["terminals"]:
            n, t = int(r["owner_uid"]), int(r["term_uid"])
            if n in scope and t not in seen:
                seen.add(t)
                names.setdefault(n, {})
                names[n][r["term_name"]] = names[n].get(r["term_name"], 0) + 1
    return dict(d, scope=scope, s1_md5_got=got, names=names)


def d4_ok(node, name, d4):
    """D4's one test: the node is in scope and S1's graph has a terminal of that name on it."""
    return node in d4["scope"] and name in d4["names"].get(node, {})


def d4_e1(diffs, info, d4):
    """(ok, accepted, bad) over [(checkpoint k, stagexec.compare dict)]; `info` = {term uid: (owner uid, name)}.
    accepted = [(k, term, node, name)] real-only terms that pass d4_ok AND stay within S1's COUNT of that name on
    that node at that checkpoint (prior-art c112d `contradicted`: review c112c-b2a-e1 :78/:91 asks names AND count;
    D4's "passes ONLY if" is a necessary condition, the count cap only narrows it); bad = every other diff entry."""
    acc, bad = [], []
    for k, d in diffs:
        bad += [(k, key, t) for key in D4_FAIL_KEYS for t in d.get(key) or []]
        seen = {}
        for t in d.get("only_real_terms") or []:
            node, name = info.get(int(t), (None, None))
            ok = d4_ok(node, name, d4) and seen.get((node, name), 0) < d4["names"][node][name]
            if ok:
                seen[(node, name)] = seen.get((node, name), 0) + 1
            (acc if ok else bad).append((k, int(t), node, name))
    return not bad, acc, bad


def _d4_terms(rows):
    out = {}
    for r in rows:
        out.setdefault(int(r["term_uid"]), (int(r["owner_uid"]), r["term_name"]))
    return out


def d4_pb(got, want, real_end, sim_end, d4):
    """(ok, closed, grown, bad) at PB. got / want = sets of (node, sink name) of cdiff(S1, real end) / the plan's
    open_rows; real_end / sim_end = terminal rows (sim uids already bound to real ones). ONE-WAY TOWARD S1: a NEW
    pair FAILS; a planned pair that CLOSED passes only on a scope node; every (node, name) a scope node holds in the
    real end beyond the simulated end (grown or renamed, by count) must pass d4_ok; every real-only terminal uid must
    pass d4_ok; a real-lost terminal uid FAILS."""
    R, S = _d4_terms(real_end), _d4_terms(sim_end)
    cr, cs = {}, {}
    for src, cnt in ((R, cr), (S, cs)):
        for v in src.values():
            if v[0] in d4["scope"]:
                cnt[v] = cnt.get(v, 0) + 1
    grown = sorted((p, c - cs.get(p, 0)) for p, c in cr.items() if c > cs.get(p, 0))
    closed = sorted(want - got)
    bad = [("new-pair", p) for p in sorted(got - want)] + [("closed-off-scope", p) for p in closed
                                                          if p[0] not in d4["scope"]]
    bad += [("grown-not-S1", p, c) for p, c in grown if not d4_ok(p[0], p[1], d4)]
    bad += [("count-over-S1", p, cr[p], d4["names"][p[0]][p[1]]) for p, _c in grown
            if d4_ok(p[0], p[1], d4) and cr[p] > d4["names"][p[0]][p[1]]]       # prior-art c112d: names AND count
    bad += [("real-only-not-S1", t, R[t]) for t in sorted(set(R) - set(S)) if not d4_ok(R[t][0], R[t][1], d4)]
    bad += [("real-lost", t, S[t]) for t in sorted(set(S) - set(R))]
    return not bad, closed, grown, bad


def d4_form(real_end, d4):
    """FACTS ONLY: {scope node: 'equal to S1' | 'real-only {..} S1-only {..}'} comparing the real end's name counts on
    each scope node with S1's (the review's stronger 'names and count' reading; never a gate here)."""
    cnt = {}
    for n, name in _d4_terms(real_end).values():
        if n in d4["scope"]:
            cnt.setdefault(n, {})
            cnt[n][name] = cnt[n].get(name, 0) + 1
    out = {}
    for n in sorted(d4["scope"]):
        a, b = cnt.get(n, {}), d4["names"].get(n, {})
        ra = dict((k, v - b.get(k, 0)) for k, v in a.items() if v > b.get(k, 0))
        sb = dict((k, v - a.get(k, 0)) for k, v in b.items() if v > a.get(k, 0))
        out[n] = "equal to S1 {0}".format(sorted(a.items())) if not (ra or sb) else \
            "real-only {0} S1-only {1}".format(sorted(ra.items()), sorted(sb.items()))
    return out

def unload_donor(path=None):
    """Card 97-5 (judgement c97): after the copy_in calls, drop the donor (default `g.MOVE_SRC`) from LabVIEW's memory
    before the MEMSTOP-metered edits. An open front panel keeps a VI resident (gscript.close_panel), so its panel is
    closed; residency is READ before/after from `Application.ExportedVIs` (NI 'All VIs in Memory'), and private MB
    before/after. FACTS ONLY: returns a dict, never raises - a failed read is recorded, not gated."""
    path = path or g.MOVE_SRC
    name = os.path.basename(path).lower()

    def resident():
        try:
            return any(os.path.basename(str(n)).lower() == name for n in (g.lv().ExportedVIs or ()))
        except Exception as e:                                                     # noqa: BLE001
            return "unread: {0}".format(str(e)[:60])
    out = {"mb_before": round((private_bytes() or 0) / 1e6, 1), "resident_before": resident()}
    try:
        g.close_panel(path)
        out["close_panel"] = "ok"
    except Exception as e:                                                         # noqa: BLE001
        out["close_panel"] = "err: {0}".format(str(e)[:80])
    time.sleep(1.0)
    out["resident_after"] = resident()
    out["mb_after"] = round((private_bytes() or 0) / 1e6, 1)
    return out


def run(fn, stage):
    """The standard main(): run `fn(stage)`, never let an exception skip the hygiene tail."""
    # card chat-T3: every real stop also prints ONE structured `STOP {json}` line (protocol.stop_line) - the line the adopt
    # checker reads; the prose lines below stay for people and for the older log readers.
    try:
        fn(stage)
    except Stop as e:
        print(_protocol.stop_line("gate", str(e)), flush=True)
        stage.gate("STOP at gate: {0}".format(e), False)
    except Exception as e:                                                         # noqa: BLE001
        import traceback
        stage.exc_text = "{0}: {1}".format(type(e).__name__, str(e))[:200]      # C6 first_fail
        stage.R["fatal"] = traceback.format_exc()[-2500:]
        print(_protocol.stop_line(_protocol.stop_kind_of(e), stage.exc_text, exc=type(e).__name__), flush=True)
        print("\nOBSERVED EXC {0}\n{1}".format(str(e)[:300], traceback.format_exc()[-2000:]), flush=True)
        stage.gate("the run completed without an unhandled exception", False, str(e)[:160])
    try:
        return stage.close()
    except Exception as e:                                                         # noqa: BLE001
        stage.gate("H-HYGIENE the hygiene phase completed", False, str(e)[:150])
        return stage.summary()

# ---------------------------------------------------------------------------------------------- PB by POSITION (card chat-T7, fp-70)
# vigraph.computation_diff (a7eb58c7, card 153-b2-1) already compares sinks and sources by position and drops a rename-only slot; a
# pred written under the older name-keyed vigraph still lists such a slot as one lost + one new NAME row ('2626|Terminal|array|0..3'
# vs S1's 'element|0..2', s06: 18 stored rows = 14 slots). PD371(b) / PD373(d): convert the stored rows to position rows - a lost +
# new pair at one (owner, index, class) is ONE changed row, a pair with equal (name-free) sources is no row - and compare SLOTS. The
# converter is the one card 153-a2-3 wrote (tools/bench/selftest_c153_poskey_convert.py:20-73, now a thin test of these functions).
# It is idempotent, so a pred already written under a7eb58c7 (P5 on, PD373(e)) passes through unchanged.
def pos_slot_table(G):
    """{term key: (node, index in the node's READ order, term class, is_source)} of a vigraph graph - the same key as
    vigraph.pos_keys / addrcheck.pos_table. Self-test: tools/bench/selftest_c153_poskey_convert.py."""
    import collections
    by = G.get("by_node")
    if by is None:
        by = collections.OrderedDict()
        for r in G["rows"].values():
            by.setdefault(r["node"], []).append(r)
    return dict((r["key"], (node, i, r.get("term_class", ""), bool(r.get("is_source")))) for node, rs in by.items()
                for i, r in enumerate(rs))


def pos_name_free(P, s):
    """One effective source ('<key>' or '<key>@<tunnel>/f<n>+...') -> (its position in table P, frame tag); an unknown key
    -> ('?', key). Self-test: tools/bench/selftest_c153_poskey_convert.py."""
    base, tag = s, ""
    if "@" in s:
        b, t = s.rsplit("@", 1)
        if all("/f" in x and x.split("/f")[0].isdigit() and x.split("/f")[-1].isdigit() for x in t.split("+")):
            base, tag = b, t
    return (P.get(base, ("?", base)), tag)


def pos_slots(G, nodes, V=None):
    """{sink key: ((node, index, class), frozenset of name-free effective sources)} for the input terminals of `nodes` in
    graph G (V = the vigraph module). Self-test: tools/bench/selftest_c153_poskey_convert.py."""
    V = V or pure("vigraph")
    P, memo, out = pos_slot_table(G), {}, {}
    for k, p in P.items():
        if not p[3] and p[0] in nodes:
            out[k] = (p[:3], frozenset(pos_name_free(P, s) for s in V.effective_sources(G, k, memo)))
    return out


def pos_convert(rows, SA, SB):
    """Name-keyed cdiff sink strings -> POSITION rows [{sink, slot, name_a, name_b, before, after[, passthrough]}], sorted by
    sink: one row per (node, index, class) whose sources differ between SA and SB (pos_slots of graph A / B); a row with no
    slot in either is passed through (slot None). Idempotent. Self-test: tools/bench/selftest_c153_poskey_convert.py S1-S7."""
    order, out = [], []
    for s in rows:
        ps = [X[s][0] for X in (SA, SB) if s in X]
        if not ps:
            out.append({"sink": s, "slot": None, "name_a": s, "name_b": s, "before": None, "after": None, "passthrough": True})
        for p in ps:
            if p not in order:
                order.append(p)
    ia = dict((v[0], (k, v[1])) for k, v in SA.items())
    ib = dict((v[0], (k, v[1])) for k, v in SB.items())
    for p in order:
        a, b = ia.get(p), ib.get(p)
        va, vb = (a[1] if a else None), (b[1] if b else None)
        if va == vb:
            continue
        out.append({"sink": a[0] if a else b[0], "slot": p, "name_a": a and a[0], "name_b": b and b[0], "before": va, "after": vb})
    return sorted(out, key=lambda r: r["sink"])


def pos_convert_open_rows(plan_open_rows, old_rows, new_rows):
    """A plan's open_rows [{node, term}] with every row dropped whose (node, name) pair the old -> new row conversion removed
    ENTIRELY (PD372(c)); the rest kept, in order. Self-test: tools/bench/selftest_c153_poskey_convert.py O1-O3."""
    nm = lambda rs: set((int(s.split("|", 1)[0]), s.split("|", 2)[2].rsplit("|", 1)[0]) for s in rs)   # noqa: E731
    gone = nm(old_rows) - nm(new_rows)
    return [y for y in plan_open_rows if (int(y["node"]), y["term"]) not in gone]

def pb_check(S1f, G1, Gpred, pred_rows, open_rows, unlicensed=(), V=None):
    """PB by POSITION (card chat-T7, fp-70): cdiff(S1f, G1 = the REAL end) (vigraph.computation_diff_frame) against the pred's cdiff
    rows and the plan's open (node, term) pairs. The pred rows are converted (pos_convert) on the slots of S1f and Gpred = the graph
    the PRED was made on (the plan's simulated end) - never on the real end, which would erase a pred row the build failed to make
    (selftest_t7_ring M2); the real rows on S1f / G1's slots. A stored NAME-keyed pred and a native position-keyed one compare the
    same way: ok = the SLOT sets are equal (a passthrough row compares by its sink string), the converted open pairs == the cdiff rows'
    (node, S1-side name) pairs, and no unlicensed rename. Sink names are report columns.
    Returns {ok, rows (cdiff sinks), pred_rows, slots_got, slots_want, extra, missing, pairs_got, pairs_want, open_rows, cd}.
    Self-test: tools/bench/selftest_t7_ring.py (stored P4 s03..s09 + P5-1a end reads, mutations)."""
    V = V or pure("vigraph")
    cd = V.computation_diff_frame(S1f, G1)
    rk = sorted(str(y["sink"]) for y in cd["rows"])
    nodes = set(int(s.split("|", 1)[0]) for s in list(pred_rows) + rk)
    SA = pos_slots(S1f, nodes, V)
    key = lambda r: tuple(r["slot"]) if r.get("slot") is not None else ("sink", r["sink"])   # noqa: E731
    want_rows, got_rows = pos_convert(list(pred_rows), SA, pos_slots(Gpred, nodes, V)), pos_convert(rk, SA, pos_slots(G1, nodes, V))
    want, got = set(key(r) for r in want_rows), set(key(r) for r in got_rows)
    op = pos_convert_open_rows(list(open_rows), list(pred_rows), [r["sink"] for r in want_rows])
    pw = set((int(y["node"]), y["term"]) for y in op)
    pg = set((int(y["node"]), str(y["sink"]).split("|")[2]) for y in cd["rows"])
    sk = lambda S: sorted(S, key=str)                                               # noqa: E731
    return {"ok": want == got and pw == pg and list(unlicensed or []) == [], "rows": rk, "pred_rows": len(pred_rows),
            "slots_got": len(got), "slots_want": len(want), "extra": sk(got - want), "missing": sk(want - got),
            "extra_sinks": sorted(r["sink"] for r in got_rows if key(r) not in want),
            "missing_sinks": sorted(r["sink"] for r in want_rows if key(r) not in got),
            "pairs_got": len(pg), "pairs_want": len(pw), "pairs_extra": sk(pg - pw), "pairs_missing": sk(pw - pg),
            "open_rows": op, "unlicensed": list(unlicensed or []), "cd": cd}

def run_bind(bind=None, report=None):
    """The run's bindings as ONE {sim uid: real uid} map: an Executor's x.bind ({obj, term, diag}) merged, plus the union of a recorded
    run's per-op `bound` dicts (x.report / a scratch json's `stagexec` list = the log's `bound {...}` lines). Returns a dict.
    Self-test: tools/bench/selftest_c158_gates386.py."""
    M = {}
    for kind in ("obj", "term", "diag"):
        for k, v in ((bind or {}).get(kind) or {}).items():
            M[_iv(k)] = _iv(v)
    for o in report or []:
        for k, v in (o.get("bound") or {}).items():
            M[_iv(k)] = _iv(v)
    return M


def real_rows_from_record(rows5, sim_terms, flat):
    """(rows, record): the 5-column real end rows a run records (RingSession.gate_td `real_end_rows`: term uid, owner uid, name, wire,
    is_source) as full terminal rows. term_class / owner_class / frame_diagram come from the sim end row of the SAME term uid, else from an
    unused sim row (negative uid) of the same bound owner, first by (name, direction), then by direction in read order; frame_diagram is
    mapped through `flat` (the AL rule of tools/bench/diag_c158_a2_3_measure.py:74-112, extracted here). record = {same_uid, by_name,
    by_order, unmatched}. Self-test: tools/bench/selftest_c158_gates386.py."""
    by_uid = dict((_iv(s["term_uid"]), s) for s in sim_terms)
    neg = {}
    for s in sim_terms:
        if _iv(s["term_uid"]) < 0:
            o = _iv(s["owner_uid"])
            neg.setdefault(flat.get(o, o), []).append(s)
    rec, pick = {"same_uid": 0, "by_name": 0, "by_order": 0, "unmatched": []}, {}
    for q in rows5:
        if _iv(q[0]) in by_uid:
            rec["same_uid"] += 1
            pick[_iv(q[0])] = by_uid[_iv(q[0])]
    for q in rows5:
        if _iv(q[0]) in pick:
            continue
        c = neg.get(_iv(q[1]), [])
        m = [s for s in c if s["term_name"] == q[2] and bool(s["is_source"]) == bool(q[4])]
        if m:
            rec["by_name"] += 1
        else:
            m = [s for s in c if bool(s["is_source"]) == bool(q[4])]
            rec["by_order"] += 1 if m else 0
        if m:
            pick[_iv(q[0])] = m[0]
            c.remove(m[0])
        else:
            rec["unmatched"].append(list(q))
    fd = lambda v: flat.get(_iv(v), _iv(v)) if v not in (None, "") else v           # noqa: E731
    rows = [{"term_uid": _iv(q[0]), "owner_uid": _iv(q[1]), "term_name": q[2], "wire_uid": _iv(q[3] or 0), "is_source": bool(q[4]),
             "term_class": pick[_iv(q[0])].get("term_class"), "owner_class": pick[_iv(q[0])].get("owner_class"),
             "frame_diagram": fd(pick[_iv(q[0])].get("frame_diagram"))} for q in rows5 if _iv(q[0]) in pick]
    return rows, rec


def frame_tag_map(G):
    """{(tunnel node uid, frame ordinal): frame diagram uid} of a FRAME-KEYED vigraph graph - what a tag '<tunnel>/f<k>' means on THAT
    graph (the ordinal is the rank of the frame uid, vigraph.py:322, so it is only meaningful inside one graph). Returns a dict.
    Self-test: tools/bench/selftest_c158_gates386.py."""
    V, out = pure("vigraph"), {}
    for k in G.get("frame_inner") or ():
        n, _c, _nm, o = V.key_parts(k)
        f = G["rows"][k].get("frame_diagram")
        if f not in (None, ""):
            out[(n, o)] = _iv(f)
    return out


def frame_src_resolve(src, fmap):
    """One frame-keyed effective source '<key>' or '<key>@<tunnel>/f<k>+...' -> (key, frozenset((tunnel uid, frame diagram uid))): every
    tag's ordinal resolved on its own graph (fmap = frame_tag_map(G)); an ordinal the map lacks stays '?f<k>'. Negative (simulated) tunnel
    uids are accepted. Self-test: tools/bench/selftest_c158_gates386.py."""
    base, tags = src, []
    if "@" in src:
        b, t = src.rsplit("@", 1)
        parts = t.split("+")
        if all("/f" in x and x.rsplit("/f", 1)[0].lstrip("-").isdigit() and x.rsplit("/f", 1)[1].isdigit() for x in parts):
            base, tags = b, parts
    out = []
    for x in tags:
        tn, o = x.rsplit("/f", 1)
        out.append((int(tn), fmap.get((int(tn), int(o)), "?f" + o)))
    return base, frozenset(out)


def case_move_frames(plan, sim_end, flat):
    """{node uid: {frame diagram uid}} - for every plan `move_in` whose dest_diagram is a frame 'new:<as>.f<k>' of a CaseStructure the SAME
    plan creates: the frame's simulated uid (sim_end['sym']) mapped through `flat` (the run's bindings). The frame a moved node's tag must
    name (PD386(a) condition (ii)). Self-test: tools/bench/selftest_c158_gates386.py."""
    acts = plan.get("actions") or []
    cases = set(a["as"] for a in acts if a.get("op") == "create" and a.get("class") == "CaseStructure" and a.get("as"))
    sym, out = sim_end.get("sym") or {}, {}
    for a in acts:
        d = a.get("dest_diagram")
        if a.get("op") != "move_in" or not isinstance(d, str) or not d.startswith("new:") or d[4:].rsplit(".f", 1)[0] not in cases:
            continue
        u = sym.get(d)
        if u is None:
            continue
        for n in a.get("nodes") or []:
            out.setdefault(_iv(n), set()).add(flat.get(_iv(u), _iv(u)))
    return out


def pb_bound_rows(cd, G, V=None):
    """{sink: frozenset of (source POSITION key, frozenset((tunnel, frame uid)))} of a frame-keyed cdiff's rows on graph G (its `after`
    side): names out (fp-70), tag ordinals resolved to frame uids. Two graphs in the SAME uid space compare equal iff their PB agrees.
    Self-test: tools/bench/selftest_c158_gates386.py."""
    V = V or pure("vigraph")
    P, fm, out = V.pos_keys(G), frame_tag_map(G), {}
    for y in cd["rows"]:
        if y["after"] is None:
            out[y["sink"]] = None
            continue
        s = set()
        for x in y["after"]:
            b, tags = frame_src_resolve(x, fm)
            s.add((P.get(b, ("?", b)), tags))
        out[y["sink"]] = frozenset(s)
    return out


def pb_check_bound(S1f, G1, Gpred, Gsimb, pred_rows, open_rows, move_frames, unlicensed=(), V=None):
    """PB of PD386(a): pb_check(S1f, G1 = the real end, Gpred, ...) PLUS (1) the real PB == the PB of Gsimb = the sim end re-keyed by the
    run's bindings (pb_bound_rows, frame uids) - any difference BLOCKS; (2) a pred-extra row of PB(S1, real) is RECORDED iff its only
    difference from S1 is the frame tag: sources untagged == S1's with the SAME source terminal uids, and every tag's frame uid is a
    plan frame (move_frames, case_move_frames) of the sink's or the source's node; every other extra / missing row BLOCKS.
    Returns {ok, old (pb_check's dict), eq_diff, recorded, blocked, pairs_extra_left, pairs_missing, n_rows}.
    Self-test: tools/bench/selftest_c158_gates386.py."""
    V = V or pure("vigraph")
    r = pb_check(S1f, G1, Gpred, pred_rows, open_rows, unlicensed, V)
    cs = V.computation_diff_frame(S1f, Gsimb)
    R_, S_ = pb_bound_rows(r["cd"], G1, V), pb_bound_rows(cs, Gsimb, V)
    eq_diff = sorted(k for k in set(R_) | set(S_) if R_.get(k) != S_.get(k))
    by, fm = dict((y["sink"], y) for y in r["cd"]["rows"]), frame_tag_map(G1)
    tu = lambda G, k: _iv((G["rows"].get(k) or {}).get("term_uid"))                # noqa: E731
    recorded, blocked = [], [{"sink": s, "why": "missing: a pred row the real end lacks"} for s in r["missing_sinks"]]
    for s in r["extra_sinks"]:
        y, why = by.get(s), None
        if y is None or y["before"] is None or y["after"] is None:
            why = "sink absent on one side"
        else:
            res = [frame_src_resolve(x, fm) for x in y["after"]]
            nodes = set([_iv(s.split("|", 1)[0])]) | set(_iv(b.split("|", 1)[0]) for b, _t in res)
            allowed = set().union(*[move_frames.get(n, set()) for n in nodes])
            tags = set(t for _b, ts in res for t in ts)
            if sorted(tu(S1f, b) for b in y["before"]) != sorted(tu(G1, b) for b, _t in res):
                why = "source terminal uids {0} != S1's {1}".format(sorted(tu(G1, b) for b, _t in res), sorted(tu(S1f, b) for b in y["before"]))
            elif not tags:
                why = "untagged difference from S1"
            elif not all(f in allowed for _tn, f in tags):
                why = "frame tag(s) {0} not the plan's frame(s) {1} for node(s) {2}".format(sorted(tags, key=str), sorted(allowed), sorted(nodes))
        (blocked if why else recorded).append({"sink": s, "why": why or "tag only: S1 source uids, plan frame", "after": y and y["after"]})
    rp = set((_iv(x["sink"].split("|", 1)[0]), x["sink"].split("|")[2]) for x in recorded)
    left = [p for p in r["pairs_extra"] if tuple(p) not in rp]
    ok = not eq_diff and not blocked and not left and not r["pairs_missing"] and list(unlicensed or []) == []
    return {"ok": ok, "old": r, "eq_diff": eq_diff, "recorded": recorded, "blocked": blocked, "pairs_extra_left": left,
            "pairs_missing": r["pairs_missing"], "n_rows": [len(R_), len(S_)]}


def pb_386(base_terms, L, real, objs, flat, plan, pred_rows, open_rows, labels, fs_pairs, renames=None, S1f=None, V=None):
    """The whole PD386(a) PB from a run's pieces - what RingSession.gate_pb calls: L = the sim end (sim uids), real = the read-back end
    rows, objs = the real objects, flat = run_bind(...). Builds G1 (pb_rename_rows, the sim end's loops re-keyed by the bindings), Gsimb
    (stagexec.bind_state of L), Gpred (L as is) and returns pb_check_bound's dict + {G1, restored, unlicensed, unbound, move_frames}.
    Self-test: tools/bench/selftest_c158_gates386.py."""
    import copy as _copy
    V, SX = V or pure("vigraph"), pure("stagexec")
    Lb, unbound = SX.bind_state(L, {"obj": flat, "term": {}, "diag": {}})
    pbr, restored, unlic = pb_rename_rows(base_terms, L["terminals"], real, renames)
    G1 = V.build4(pbr, objs, _copy.deepcopy(Lb["loops"]), labels, fs_pairs, frame_keyed=True)
    mf = case_move_frames(plan, L, flat)
    r = pb_check_bound(S1f or s1_frame_graph(labels), G1, pred_end_graph(L, labels, fs_pairs), pred_end_graph(Lb, labels, fs_pairs),
                       pred_rows, open_rows, mf, unlic, V)
    r.update({"G1": G1, "restored": restored, "unlicensed": unlic, "unbound": sorted(unbound),
              "move_frames": dict((n, sorted(f)) for n, f in mf.items())})
    return r


def wire_pairs_pos(rows, ob=None, tb=None, base_nodes=None):
    """(Counter of (source end, sink end) pairs, {wire uid: sorted source ends} of source-only stubs) of one terminal table, ends = position
    keys (_pos_rows, nodes mapped through ob / tb). An end on a node NOT in base_nodes (a created object) drops its index (-1) when its
    (node, class, direction) is unique on that node: the simulator's terminal order on a created object is not LabVIEW's (kept_nodes;
    the created shift register 31347's two faces swap index 0/1, diag_c158_a2_3_facts.md:171-172). A wire with sinks but no source pairs
    as (None, sink). Self-test: tools/bench/selftest_c158_gates386.py."""
    import collections as _co
    P = _pos_rows(rows, ob, tb)
    grp = _co.Counter((k[0], k[2], k[3]) for k in P)
    bn = base_nodes
    end = lambda k: (k[0], -1, k[2], k[3]) if bn is not None and k[0] not in bn and grp[(k[0], k[2], k[3])] == 1 else k   # noqa: E731
    pairs, stubs = _co.Counter(), {}
    for w, (a, b) in wire_ends_pos(rows, ob, tb).items():
        if not b:
            stubs[w] = sorted(end(x) for x in a)
            continue
        for x in (sorted(a) or [None]):
            for y in sorted(b):
                pairs[(end(x) if x is not None else None, end(y))] += 1
    return pairs, stubs


def term_identity_pairs(base, sim, real, bind=None, renames=None):
    """D / TD of PD386(b): term_identity_pos's dict with d_ok / td_ok re-judged PAIR-KEYED. D: the new and the lost (source, sink) pairs
    (wire_pairs_pos, sim owners via `bind`) of real == the sim's, as multisets, else BLOCK; source-only stubs are dropped on both sides
    and RECORDED (`stubs` sim_only / real_only / both, by source ends). TD: an `unwired` base key whose wire in the sim end is a
    source-only stub is RECORDED (`unwired_recorded`); moved, lost != gone, or any other unwired key BLOCKS. Adds `pairs` (counts,
    one-sided pairs) and `d_key`. Self-test: tools/bench/selftest_c158_gates386.py."""
    ti = term_identity_pos(base, sim, real, bind=bind, renames=renames)
    tb, ob = _bind_maps(bind)
    bn = set(k[0] for k in _pos_rows(base))
    Pb, _Hb = wire_pairs_pos(base, base_nodes=bn)
    Ps, Hs = wire_pairs_pos(sim, ob, tb, bn)
    Pr, Hr = wire_pairs_pos(real, base_nodes=bn)
    ns, nr, ls, lr = Ps - Pb, Pr - Pb, Pb - Ps, Pb - Pr
    d_ok = ns == nr and ls == lr
    hs, hr = dict((w, tuple(v)) for w, v in Hs.items()), dict((w, tuple(v)) for w, v in Hr.items())
    vs, vr = set(hs.values()), set(hr.values())
    stubs = {"sim_only": [{"wire": w, "sources": list(v)} for w, v in sorted(hs.items(), key=str) if v not in vr],
             "real_only": [{"wire": w, "sources": list(v)} for w, v in sorted(hr.items(), key=str) if v not in vs],
             "both": len(vs & vr)}
    S = _pos_rows(sim)
    rec = [k for k in ti["unwired"] if k in S and _iv(S[k].get("wire_uid") or 0) in Hs]
    unw = [k for k in ti["unwired"] if k not in rec]
    st = lambda C: sorted(map(str, C.elements()))[:12]                              # noqa: E731
    out = dict(ti)
    out.update({"d_ok": d_ok, "td_ok": not unw and not ti["moved"] and set(ti["lost"]) == set(ti["gone"]), "unwired": unw,
                "unwired_recorded": rec, "stubs": stubs, "d_key": "(source, sink) position pairs; source-only stubs RECORDED (PD386(b))",
                "pairs": {"new_real": sum(nr.values()), "new_sim": sum(ns.values()), "lost_real": sum(lr.values()), "lost_sim": sum(ls.values()),
                          "new_only_real": st(nr - ns), "new_only_sim": st(ns - nr), "lost_only_real": st(lr - ls), "lost_only_sim": st(ls - lr)}})
    return out


def scratch_census(log_path, sum_json=None, key="CENSUS-ALL new uids by class (every class) "):
    """(census {class: n}, whole-file line number, last-segment start line) from the ONE `key` line in the LAST bgrun segment of a
    scratch log (tools/bench/diag_c148_a2_4_census.py:31-40 as a call); ValueError when there is not exactly one such line, or when
    sum_json's census_all.new_by_class differs. Self-test: tools/bench/selftest_c158_gates386.py."""
    lines = open(log_path, encoding="utf-8", errors="replace").read().splitlines()
    seg0 = max([i for i, ln in enumerate(lines) if ln.startswith("BGRUN START")] or [0])
    hit = [(i + 1, ln) for i, ln in enumerate(lines) if i >= seg0 and key in ln]
    if len(hit) != 1:
        raise ValueError("scratch_census: {0} '{1}' lines in the last segment of {2}".format(len(hit), key.strip(), log_path))
    meas = json.loads(hit[0][1].split(key, 1)[1])
    if sum_json:
        sj = (jload(sum_json).get("census_all") or {}).get("new_by_class")
        if sj != meas:
            raise ValueError("scratch_census: log line {0} != {1} census_all.new_by_class {2}".format(meas, sum_json, sj))
    return meas, hit[0][0], seg0 + 1


def op_class_census(prev_rows, rows):
    """Class census between two whole-VI terminal reads (no extra LabVIEW read): {terms: {term class: n new term uids}, owners: {owner
    class: n new owner uids}, wires: n new wire uids, lost_terms, lost_owners, lost_wires}. Self-test: tools/bench/selftest_c158_gates386.py."""
    def idx(rs):
        t, o, w = {}, {}, set()
        for r in rs or []:
            t[_iv(r["term_uid"])] = r.get("term_class")
            o[_iv(r["owner_uid"])] = r.get("owner_class")
            if _on(_iv(r.get("wire_uid") or 0)):
                w.add(_iv(r["wire_uid"]))
        return t, o, w
    (ta, oa, wa), (tb_, ob_, wb) = idx(prev_rows), idx(rows)
    cnt = lambda d, ks: dict(sorted(__import__("collections").Counter(str(d[k]) for k in ks).items()))   # noqa: E731
    return {"terms": cnt(tb_, set(tb_) - set(ta)), "owners": cnt(ob_, set(ob_) - set(oa)), "wires": len(wb - wa),
            "lost_terms": len(set(ta) - set(tb_)), "lost_owners": len(set(oa) - set(ob_)), "lost_wires": len(wa - wb)}


def op_class_census_install(rs, ops=(16, 17)):
    """PRINTING ONLY (PD386(c), the PD385(b) per-op census owed): wrap the ring backend's read (sink_gate_backend_class) so the whole-VI
    terminal reads taken after ops k-1 and k (k in `ops`; the op = len(rs.x.report) + 1 at read time) are KEPT - no extra read, no act
    change. op_class_census_print prints them. Returns the holder dict. Self-test: tools/bench/selftest_c158_gates386.py."""
    cls = sink_gate_backend_class()
    want = set(ops) | set(k - 1 for k in ops)
    H = {"rs": rs, "ops": tuple(ops), "want": want, "reads": {}}
    if not getattr(cls, "_opc_base", None):
        cls._opc_base = cls.read

        def read(self):
            rows = cls._opc_base(self)
            h = getattr(cls, "_opc", None)
            x = h and getattr(h["rs"], "x", None)
            if x is not None:
                k = len(getattr(x, "report", None) or []) + 1
                if k in h["want"]:
                    h["reads"][k] = list(rows or [])
            return rows
        cls.read = read
    cls._opc = H
    return H


def op_class_census_print(h):
    """Print one `OPCENSUS op <k>` line per op of an op_class_census_install holder (or why it was not measured). Returns the dict."""
    out = {}
    for k in h["ops"]:
        a, b = h["reads"].get(k - 1), h["reads"].get(k)
        if a is None or b is None:
            print("  OPCENSUS op {0} not measured (whole read after op {1}: {2}, after op {0}: {3}; dry run or op not reached)".format(
                k, k - 1, a is not None, b is not None), flush=True)
            continue
        out[k] = op_class_census(a, b)
        print("  OPCENSUS op {0} (terminal reads after op {1} -> op {0}, no extra read) {2}".format(k, k - 1, json.dumps(out[k], sort_keys=True)),
              flush=True)
    return out


def retry_card_arg(argv, script):
    """PRINTING ONLY: the `--retry-card <path>` / `RETRY_CARD=<path>` judgement card in a recipe's own argv (stage_prerun.RETRY_CARD_RE,
    the form the launch gate reads) -> {path, id, retry_of, matches: retry_of == stage_prerun.stage_key(script)}; {'path': None} when the
    card rides on the bgrun line instead (bgrun / the launch gate read it there). Never refuses. Self-test: selftest_c158_gates386.py."""
    SP = pure("stage_prerun")
    m = SP.RETRY_CARD_RE.search(" ".join(argv or []))
    if not m:
        return {"path": None, "note": "not in this process's argv (a bgrun --retry-card is read by the launch gate and bgrun)"}
    p = m.group(1) if os.path.isabs(m.group(1)) else os.path.join(ROOT, m.group(1))
    try:
        c = jload(p)
    except (OSError, ValueError) as e:
        return {"path": m.group(1), "error": str(e)[:160]}
    return {"path": m.group(1), "id": c.get("id"), "retry_of": c.get("retry_of"), "matches": c.get("retry_of") == SP.stage_key(script)}

def sink_gate_spec(e, actions, base, sim_end):
    """One sink-gate census row (prep_*_session.json `sink_gate_census`, needs_sink_gate) -> its declared sink gate spec
    {n, row, gate, acts, terms, sink, census_sink, src_name}: acts = the plan's wire row, terms = its base dst term(s), sink =
    (owner, name) from the base graph, src_name = the source terminal's name (base for a numeric source, the simulated end for a
    symbolic 'new:' source). As stage_d1_ring_p5s01a.py:56-75 (_sg_spec). Self-test: tools/bench/selftest_t7_ring.py."""
    import re as _re
    acts = [a for a in actions if a.get("id") == e["id"] and a["op"] == "wire" and isinstance(a.get("dst"), dict)]   # a symbolic dst is a str
    terms = sorted(set(a["dst"]["term_uid"] for a in acts))
    sinks = sorted(set((r["owner_uid"], r["term_name"]) for r in base["terminals"] if r["term_uid"] in terms))
    src = acts[0]["src"] if acts else None
    if isinstance(src, dict):
        sname = sorted(set(r["term_name"] for r in base["terminals"] if r["term_uid"] == src["term_uid"]))
    elif isinstance(src, str) and src.startswith("new:"):
        try:
            sname = [pure("stagesim").resolve_addr(sim_end, src, True)["term_name"]]
        except Exception as ex:                                                    # noqa: BLE001 - the SG gate reports it
            sname = ["UNRESOLVED: {0!r}".format(ex)]
    else:
        sname = []
    sink = list(sinks[0]) if len(sinks) == 1 else None                             # the sink key exactly as the run reports it
    slug = _re.sub(r"[^a-z0-9]+", "-", str(sink[1] if sink else "").lower()).strip("-")
    return {"n": e["n"], "row": e["id"], "gate": "SG-{0}-{1}".format(e["id"], slug), "acts": acts, "terms": terms, "sink": sink,
            "census_sink": list(e["sink"]), "src_name": sname}


def sink_gate_read(e, d, x, rows, n_actions):
    """(ok, detail) for the deferred op error on sink d['sink'] (stage_d1_ring_p5s01a.py:90-117 sg_read, unchanged): (a) the indicator
    terminal exists with exactly ONE wire, (b) that wire is the net of the plan source (through x.bind; a symbolic source resolved
    on the simulated end x.step(n_actions)), (c) that net has exactly ONE source = that terminal, (d) `rows` are the executor's
    whole-VI read after its LAST op. x = the stagexec.Executor, rows = the backend's last read. Self-test: selftest_t7_ring.py."""
    a = d["acts"][0]
    tb, ob = (lambda t: x.bind["term"].get(t, t)), (lambda u: x.bind["obj"].get(u, u))
    if isinstance(a["src"], dict):
        ts, so = tb(a["src"]["term_uid"]), ob(a["src"]["uid"])
    else:
        sr = pure("stagesim").resolve_addr(x.step(n_actions)["state"], a["src"], True)
        ts, so = tb(sr["term_uid"]), ob(sr["owner_uid"])
    td = tb(a["dst"]["term_uid"])
    rows = rows or []
    d_ok = bool(rows) and bool(x.reads_real) and x.reads_real[-1] == len(x.ops)
    ind = sorted(set((r["term_uid"], r["owner_uid"], r["term_name"], r["wire_uid"]) for r in rows if r["term_uid"] == td))
    src = sorted(set((r["term_uid"], r["owner_uid"], r["term_name"], r["wire_uid"]) for r in rows if r["term_uid"] == ts))
    w = ind[0][3] if len(ind) == 1 else None
    a_ok = len(ind) == 1 and bool(w) and [ind[0][1], ind[0][2]] == list(e.get("sink") or [])
    b_ok = (bool(w) and len(src) == 1 and src[0][3] == w and src[0][1] == so and [src[0][2]] == d["src_name"]
            and list(e.get("source") or []) == [so, d["src_name"][0]])
    srcs = sorted(set(r["term_uid"] for r in rows if w and r["wire_uid"] == w and r["is_source"]))
    c_ok = srcs == [ts]
    return a_ok and b_ok and c_ok and d_ok, ("(a) indicator t{0} rows {1} wire {2}: {3}; (b) source t{4} #{5} rows {6} wire(s) {7} entry source {8}: {9}; "
                                             "(c) sources on wire {2} {10} == [t{4}]: {11}; (d) rows {12} = read after op {13} of {14}: {15}").format(
        td, ind, w, a_ok, ts, so, src, sorted(set(y[3] for y in src)), e.get("source"), b_ok, srcs, c_ok,
        len(rows), (x.reads_real or [None])[-1], len(x.ops), d_ok)

class SimCensus(dict):
    """card 159-a4-1 (PD387(e)): a {uid: class} census snapshot taken from a stagesim STATE, not from LabVIEW; Stage.census_gate
    compares two of them even in a dry run. Self-test: tools/bench/selftest_c159_a4_1.py."""
    source = "stagesim"


def sim_census_snapshot(state, classes=None):
    """card 159-a4-1 (PD387(e)): the SimCensus {uid: class} of one stagesim state - objs by class, terminals by term_class, every
    wire uid on a terminal row as 'Wire', every diagram uid as 'Diagram' - kept to `classes` when given (as RingSession.census_snapshot
    keeps the pred's classes). Returns a SimCensus. Self-test: tools/bench/selftest_c159_a4_1.py."""
    d = SimCensus()
    for o in state.get("objs") or []:
        d[int(o["uid"])] = str(o["class"])
    for r in state.get("terminals") or []:
        d[int(r["term_uid"])] = str(r.get("term_class") or "Terminal")
        if r.get("wire_uid"):
            d[int(r["wire_uid"])] = "Wire"
    for k in state.get("diagrams") or {}:
        d.setdefault(int(k), "Diagram")
    if classes is not None:
        keep = set(classes)
        for u in [u for u, c in d.items() if c not in keep]:
            del d[u]
    return d


def census_unpredicted_reason(pred):
    """card 159-a4-1 (PD387(e), docs/violation-decisions.md device-failed 2026-10-10 03:2x): None when a pred's census block covers
    every op of its plan, else why it is UNPREDICTED - an empty census, census_overall UNPREDICTED or absent, or any non-measured
    verdict with rows left in census_unpredicted. A MEASURED-* census (one whole-run measurement) covers every op. Returns None or a
    str. Self-test: tools/bench/selftest_c159_a4_1.py."""
    pred = pred or {}
    un = list(pred.get("census_unpredicted") or [])
    ov = str(pred.get("census_overall") or "")
    if not pred.get("census"):
        return "census block empty"
    if ov.startswith("MEASURED"):
        return None
    if not ov:
        return "census_overall absent - coverage of the plan's ops not stated"
    if ov == "UNPREDICTED" or un:
        return "census_overall {0}, {1} plan row(s) not predicted {2}".format(ov, len(un), un[:6])
    return None if ov == "PASS" else "census_overall {0}".format(ov)


class RingSession(object):
    """The inputs of one ring stage session and the gates its body runs - stage_d1_ring_p5s01a.py:31-191 as calls. A recipe builds
    ONE RingSession at module level from its four input files and calls, in body(s): gate_l0 -> start/discard_work -> handles ->
    census_snapshot -> executor -> run -> gates_ng_fr_d -> gate_cen2 (with a second census_snapshot) -> gate_td -> s.es -> gate_pb
    -> finish. The two census_snapshot CALL SITES stay in the recipe source on purpose: stage_prerun's X10 counts whole-VI reads from
    the recipe source by call name (stage_prerun.py:2191 X10_SRC_READS). Self-test: tools/bench/selftest_t7_ring.py."""

    def __init__(self, plan, pred, licence, census):
        SX = pure("stagexec")
        self.plan_path, self.pred_path, self.licence_path, self.census_path = plan, pred, licence, census
        self.P = jload(plan)
        self.BASE = jload(ROOT, self.P["finalized"]["base"]["path"])
        self.A = self.P["actions"]
        self.PRED = jload(pred)
        self.REN = licence_renames(jload(licence))                                 # PD365(d): only echoed since PD366(f)
        self.NC = sum(1 for a in self.A if a["op"] == "create" and a.get("as"))   # created objects, from the plan (FR gate)
        self.DRY = bool(getattr(g.report_all, "_dry", False))
        self.LAB = pure("jev_candidates").node_labels_default()
        self.SRC = tuple(self.PRED["census"])
        ops = SX.compile_plan(self.P)
        self.BIND = set(k for k, o in enumerate(ops, 1) if o["kind"] in SX.BIND_KINDS)   # PD265(c): every op that creates/binds
        self.CHECKPOINTS = SX.checkpoint_ops(ops)   # card 157-a1-1 (PD381(d)): {0, last} | BIND | every move_in op
        self.FSPAIRS = self.BASE.get("fs_tunnel_pairs", self.BASE.get("fs_pairs"))
        self.SG_CENSUS = [e for e in jload(census).get("sink_gate_census") or [] if e.get("needs_sink_gate")]   # PD366(e)
        self.SIM_END = jload(ROOT, self.P["finalized"]["step_files"][-1]["path"])["state"]
        self.SGS = [sink_gate_spec(e, self.A, self.BASE, self.SIM_END) for e in self.SG_CENSUS]
        self.SINK_GATES = [{"gate": d["gate"], "sink": d["sink"]} for d in self.SGS]
        self.SINK_FNS = dict((d["gate"], (lambda e, _d=d: self.sink_read(e, _d))) for d in self.SGS)
        self.x = self.be = self.L = self.ti = None
        self.made, self.on = [], []

    def stem(self):
        """The plan file's name without .json (gate labels name it). Returns a str."""
        return os.path.splitext(os.path.basename(self.plan_path))[0]

    def sink_read(self, e, d):
        """(ok, detail) of sink_gate_read on this session's executor and last read; (True, 'dry') in a dry run."""
        if self.DRY:
            return True, "dry"
        return sink_gate_read(e, d, self.x, getattr(self.be, "last_terms", None), len(self.A))

    def census_snapshot(self, s):
        """{uid: class} of s.census_snapshot() kept to the pred's census classes (the recipes' `snap`). ONE whole-VI read."""
        return dict((u, c) for u, c in s.census_snapshot().items() if c in self.SRC)

    def handles(self):
        """LabVIEW's handle count now (bench_prep.labview_handles); None in a dry run is the caller's business."""
        return mod("bench_prep").labview_handles()

    def gate_l0(self, s):
        """L0 (FATAL): plan final + open_rows_match + plan_in == <plan>_in; base graph md5 == pred == file; bed md5 == pred; plan end
        cdiff == pred; pred keyed to this plan's md5."""
        P, PRED, fz = self.P, self.PRED, self.P["finalized"]
        st = self.stem()
        s.gate("L0 plan FINAL, open_rows_match, plan_in = {0}_in; base graph md5 == pred == file; bed md5 == pred; plan end cdiff == pred; pred keyed to this plan".format(st),
               P.get("final") is True and fz.get("open_rows_match") is True and os.path.splitext(fz["plan_in"]["path"])[0].endswith(st + "_in")
               and fz["base"]["md5"] == PRED["graph"]["md5"] == md5(os.path.join(ROOT, fz["base"]["path"])) and self.BASE["md5"] == PRED["bed_md5"]
               and sorted(fz["end_cdiff_rows"]) == PRED["cdiff_rows"] and PRED["plan"]["md5"] == md5(self.plan_path),
               (md5(self.plan_path), fz["base"], self.BASE["md5"]), fatal=True)

    def executor(self, s):
        """SG (FATAL: one declared sink gate per census row) -> the backend (stagexec.DryPlanBE in a dry run, else the SGBackend
        with the sink gates) -> stagexec.Executor (checkpoints, name gate) -> L1 (FATAL: actions compile into the pred's op kinds,
        each action once). Returns the Executor (also self.x; the backend is self.be)."""
        SX = pure("stagexec")
        sg = SX.check_sink_gates(self.SINK_GATES, self.SINK_FNS)                   # card 147-a2-1: the same check LVBackend makes
        SGS = self.SGS
        s.gate("SG one declared sink gate per census row ({0} rows of {1}): {2}; per row: sink term(s), plan action(s), source, source name {3}".format(
            len(self.SG_CENSUS), os.path.basename(self.census_path), sg, [(d["row"], d["terms"], [a["id"] for a in d["acts"]], [a["src"] for a in d["acts"]], d["src_name"]) for d in SGS]),
               len(sg) == len(self.SG_CENSUS) == len(SGS) and len(set(d["gate"] for d in SGS)) == len(SGS)
               and all(sg[k]["sink"] == d["sink"] == d["census_sink"] and len(d["terms"]) == 1 and len(d["acts"]) == 1 and len(d["src_name"]) == 1
                       and not str(d["src_name"][0]).startswith("UNRESOLVED") for k, d in enumerate(SGS)), sg, fatal=True)
        self.be = SX.DryPlanBE(s, self.P, self.plan_path, self.BASE) if self.DRY else sink_gate_backend_class()(
            s, self.FSPAIRS, sink_gates=self.SINK_GATES, gates=self.SINK_FNS, mem_stop_mb=SX.MEM_STOP_MB)
        self.x = SX.Executor(self.plan_path, self.be, log=lambda m: print(m, flush=True), checkpoints=self.CHECKPOINTS, name_gate=True)
        x, A = self.x, self.A
        s.gate("L1 the {0} actions compile into {1} real ops == pred kinds, each action once; checkpoints {2}".format(len(A), len(x.ops), self.CHECKPOINTS),
               [o["kind"] for o in x.ops] == self.PRED["ops"] and sorted(n for o in x.ops for n in o["acts"]) == list(range(1, len(A) + 1)),
               [o["kind"] for o in x.ops], fatal=True)
        return x

    def run(self, s):
        """Run the executor: E1 (every checkpoint's real graph == its simulated step). On stagexec.ExecStop: save ops 1..k-1 for a
        same-cycle resume (PD329(b)), write the stop report, return None. Returns the real end rows otherwise."""
        SX, x = pure("stagexec"), self.x
        try:
            real = x.run()
            s.gate("E1 every checkpoint's real graph == its simulated step ({0} ops)".format(len(x.ops)), True)
        except SX.ExecStop as e:
            try:
                SX.save_for_resume(s, x, e)
            except Exception as ee:                                                # noqa: BLE001 - the stop report must still be written
                s.fact("RESUME save failed: {0!r}".format(ee))
            SX.report_stop(s, x, self.be, e)
            return None
        s.R["stagexec"] = x.report
        return real

    def gates_ng_fr_d(self, s, real):
        """NG (crossing ops' new tunnel names == the simulator's), FR (FATAL: created objects on the sim end's frames; real frames ==
        the sim end's), D (new / lost wires == the sim's, position keys via term_identity_gates). Returns the term_identity dict."""
        x, A, L = self.x, self.A, None
        frames = lambda rows: set(int(r["frame_diagram"] or 0) for r in rows)      # noqa: E731
        s.gate("NG every crossing op's NEW tunnel names == the simulator's (stagexec.tunnel_name_check, per op, from the step files): {0} op(s) {1}".format(
            len(x.name_checks), [c["k"] for c in x.name_checks]), all(c["ok"] for c in x.name_checks), x.name_checks[:6])
        L = self.L = x.step(len(A))["state"]
        rb = lambda sym: x.bind["obj"].get(L["sym"][sym], x.bind["diag"].get(L["sym"][sym], L["sym"][sym]))   # noqa: E731
        made = self.made = [rb("new:" + a["as"]) for a in A if a["op"] == "create" and a.get("as")]
        on = self.on = sorted(set(int(r["frame_diagram"] or 0) for r in real if r["owner_uid"] in made))
        sf = set(int(x.bind["diag"].get(f, f)) for f in frames(L["terminals"]))   # the simulated end's frames, through the diagram binding
        NC = self.NC
        s.gate("FR the {0} created objects sit on {1} frame(s) of the simulated end; real frames == the sim end's".format(len(made), on),
               len(made) == NC and (on or NC == 0) and set(on) <= sf and frames(real) == sf,
               {"on": on, "nc": NC, "extra": sorted(frames(real) - sf), "missing": sorted(sf - frames(real))}, fatal=True)
        # card 158-a3-1 (PD386(b)): D is judged PAIR-KEYED (term_identity_pairs); source-only stubs are RECORDED, not compared
        ti = self.ti = term_identity_pairs(self.BASE["terminals"], L["terminals"], real, renames=self.REN, bind={"term": x.bind["term"], "obj": x.bind["obj"]})
        pz = ti["pairs"]
        s.gate("D new (source, sink) pairs {0} == sim's {1}; lost {2} == sim's {3} (position pairs, stagekit.term_identity_pairs PD386(b)); source-only stubs RECORDED sim-only {4} real-only {5}; deleted uids re-used: noted {6}".format(
            pz["new_real"], pz["new_sim"], pz["lost_real"], pz["lost_sim"], [d["wire"] for d in ti["stubs"]["sim_only"]], [d["wire"] for d in ti["stubs"]["real_only"]],
            [n["wire"] for n in ti["reuse_noted"]]), ti["d_ok"], {"pairs": pz, "wires_new": ti["new"], "wires_lost": ti["lost_w"][:20], "reuse_noted": ti["reuse_noted"][:6]})
        for d in ti["stubs"]["sim_only"] + ti["stubs"]["real_only"]:
            s.fact("RECORD D source-only stub (no sink, carries nothing; PD386(b)) wire {0} sources {1} - {2} only".format(
                d["wire"], d["sources"], "sim" if d in ti["stubs"]["sim_only"] else "real"))
        return ti

    def gate_cen2(self, s, c0, c1):
        """CEN2: the new-object census between snapshots c0 and c1 == the pred's census. Returns s.census_gate's delta.
        card 159-a4-1 (PD387(e)): in a DRY run c0 / c1 are the SIMULATED census of the plan's step_00 and of its simulated end
        (sim_census_snapshot, the pred's classes), so CEN2 is computed, not stubbed; a census that does not cover every op of the
        plan (census_unpredicted_reason) is UNPREDICTED, never a PASS."""
        label = "CEN2 new-object census (classes of the pred) == {0} census".format(os.path.basename(self.pred_path))
        if self.DRY:
            c0 = sim_census_snapshot(jload(ROOT, self.P["finalized"]["step_files"][0]["path"])["state"], self.SRC)
            c1 = sim_census_snapshot(self.SIM_END, self.SRC)
        why = census_unpredicted_reason(self.PRED)
        if why:
            return s.census_gate(label, c0, c1, self.PRED["census"], unpredicted=why)
        return s.census_gate(label, c0, c1, self.PRED["census"])

    def gate_td(self, s, real):
        """TD: every base terminal the sim keeps wired is still wired; identities lost == the plan's deletes (position key). Records
        s.R['term_identity'] and s.R['real_end_rows']."""
        ti = self.ti
        s.gate("TD every base terminal the sim keeps wired is still wired; identities lost == the plan's deletes (key {0}, stagekit.term_identity_gates); renames {1}, unlicensed {2}".format(
            ti.get("td_key", "uid/owner/name"), len(ti.get("renames") or []), ti.get("unlicensed_renames")),
               ti["td_ok"], {"unwired": ti["unwired"][:20], "lost": ti["lost"][:20], "plan_deletes": ti["gone"][:20], "recycled_uids": ti["recycled"][:20],
                             "raw_lost": ti["raw_lost"][:20], "renames": (ti.get("renames") or [])[:10], "unlicensed_renames": ti.get("unlicensed_renames"),
                             "unwired_recorded": ti.get("unwired_recorded")})
        for k in ti.get("unwired_recorded") or []:                                 # card 158-a3-1 (PD386(b))
            s.fact("RECORD TD base key {0} unwired in real: its sim-end wire is a source-only stub (no sink; PD386(b))".format(k))
        s.R["term_identity"] = ti
        s.R["real_end_rows"] = [[r["term_uid"], r.get("owner_uid"), r.get("term_name"), r.get("wire_uid"), bool(r.get("is_source"))] for r in real]   # review c141-2 s4.2

    def gate_pb(self, s, real):
        """PB (FATAL, before save): pb_check(S1, real end) - by POSITION (fp-70). The real end's rows get the sim end's names on kept
        base nodes first (pb_rename_rows, PD366(e)); objects = the backend's last read. Returns the cdiff sink list."""
        # card 158-a3-1 (PD386(a)): pb_386 - real PB == the sim end's PB re-keyed by the run's bindings (frame uids, not ordinals); a
        # pred-extra row that differs from S1 only by a plan-frame tag with S1's source terminal uids is RECORDED; anything else BLOCKS
        L, be = self.L, self.be
        r = pb_386(self.BASE["terminals"], L, real, getattr(be, "last_objs", None) or be.st["objs"], run_bind(bind=self.x.bind), self.P,
                   self.PRED["cdiff_rows"], self.P["open_rows"], self.LAB, self.FSPAIRS, self.REN)
        o = r["old"]
        for x in r["recorded"]:
            s.fact("RECORD PB {0}: {1} (PD386(a)) - after {2}".format(x["sink"], x["why"], x["after"]))
        s.gate("PB frame-keyed cdiff(S1, real end) by POSITION (owner, index, class) == the sim end's re-keyed by the run's bindings ({0} rows) and == the pred's {1} rows ({2} slots) + {3} RECORDED tag-only row(s), the plan's {4} open (node, term) pairs; no unlicensed rename (FATAL, before save)".format(
            r["n_rows"], o["pred_rows"], o["slots_want"], len(r["recorded"]), len(self.P["open_rows"])), r["ok"],
               {"eq_diff": r["eq_diff"][:6], "blocked": r["blocked"][:6], "pairs_extra_left": r["pairs_extra_left"], "pairs_missing": r["pairs_missing"],
                "recorded": [x["sink"] for x in r["recorded"]], "move_frames": r["move_frames"], "restored": r["restored"], "unlicensed": r["unlicensed"]}, fatal=True)
        return o["rows"]

    def finish(self, s, h0, rk, dc, key, level, save=True):
        """PH / HB (handles open -> save <= +700, PD236(b)), the rule-6 GUI save (broken by design), PS (saved, md5 differs, input
        unchanged), the work copy kept, and s.R[key] = {final, md5, cdiff_rows, handles, census_delta, frames, errorlist totals,
        level}; s.dump(). Returns the saved md5 (None when not saved)."""
        h1 = None if self.DRY else self.handles()
        s.fact("PH handles: post-open {0} -> before-save {1}".format(h0, h1))
        s.gate("HB open->save handle growth <= +700 (PD236(b) band; refs balance is H5)", self.DRY or (h1 is not None and h0 is not None and h1 - h0 <= 700), (h0, h1))
        m = s.save(broken_ok=True) if save else None
        if save:
            s.gate("PS artefact saved, md5 differs from the input; input unchanged", m and m != s.input_md5 and md5(s.input_vi) == s.input_md5, m)
        if m and m != s.input_md5 and s.work in s.scratches:
            s.scratches.remove(s.work)
        el = self.PRED["errorlist"]                                                 # in-between file: checked False, no EL file written
        s.R[key] = {"final": s.work if m else None, "md5": m, "cdiff_rows": rk, "handles": [h0, h1], "census_delta": dc, "frames": self.on,
                    "errorlist_predicted_total": el["predicted_total"], "errorlist_alternative_total": el["alternative_total"], "level": level}
        s.dump()
        return m

def pred_x10(pred_path, plan_path):
    """The X10 prediction of a plan: its pred file's memory_pred.peak_mb, ONLY when pred.plan.md5 == md5(plan) (the scratch
    recipes' x10()). Returns a float or None. Self-test: tools/bench/selftest_t7_ring.py."""
    try:
        pr = jload(pred_path)
        if (pr.get("plan") or {}).get("md5") != md5(plan_path):
            return None
        v = (pr.get("memory_pred") or {}).get("peak_mb")
        return float(v) if v is not None else None
    except (OSError, ValueError):
        return None


_CENSUS_ALL = {"snaps": [], "on": False}


def census_all_install(stage_cls=None):
    """Wrap Stage.census_snapshot / census_gate (once) so every snapshot is kept and every census gate also prints the CENSUS-ALL
    lines (net delta / new uids by class / lost uids, all classes) and stores s.R['census_all'] - the scratch recipes' snap_keep /
    cg_print. Returns the dict of kept snapshots. Self-test: tools/bench/selftest_t7_ring.py."""
    cls = stage_cls or Stage
    if getattr(cls, "_census_all", False):
        return _CENSUS_ALL
    _snap, _cg = cls.census_snapshot, cls.census_gate

    def snap_keep(self, target=None):
        d = _snap(self, target)
        _CENSUS_ALL["snaps"].append(dict(d))
        return d

    def cg_print(self, label, before, after, declared, **kw):
        got = _cg(self, label, before, after, declared, **kw)                       # card 159-a4-1: unpredicted= passes through
        if isinstance(before, dict) and isinstance(after, dict) and len(_CENSUS_ALL["snaps"]) >= 2:
            b, a = _CENSUS_ALL["snaps"][0], _CENSUS_ALL["snaps"][-1]
            cnt = lambda d: dict((c, sum(1 for v in d.values() if v == c)) for c in set(d.values()))   # noqa: E731
            cb, ca = cnt(b), cnt(a)
            delta = dict((c, ca.get(c, 0) - cb.get(c, 0)) for c in sorted(set(cb) | set(ca)) if ca.get(c, 0) != cb.get(c, 0))
            nb = {}
            for u in set(a) - set(b):
                nb[a[u]] = nb.get(a[u], 0) + 1
            print("  CENSUS-ALL net delta (every class) {0}".format(json.dumps(delta, sort_keys=True)), flush=True)
            print("  CENSUS-ALL new uids by class (every class) {0}".format(json.dumps(nb, sort_keys=True)), flush=True)
            print("  CENSUS-ALL lost uids {0}".format(len(set(b) - set(a))), flush=True)
            self.R["census_all"] = {"net_delta": delta, "new_by_class": nb, "lost": len(set(b) - set(a))}
        return got
    cls.census_snapshot, cls.census_gate, cls._census_all = snap_keep, cg_print, True
    return _CENSUS_ALL


EL_OCR_ALIASES = (("vl", "vi"), ("v", "y"))                                         # == errorlist_check.OCR_ALIASES
EL_TOTAL_TOL = 3                                                                    # PD354(a): RETURN when |total diff| > 3


def el_norm_ocr(s):
    """An Error List text -> lower-case alphanumerics with the OCR aliases applied (== errorlist_check.norm). Returns a str."""
    import re as _re
    k = _re.sub(r"[^a-z0-9]", "", (s or "").lower())
    for a, b in EL_OCR_ALIASES:
        k = k.replace(a, b)
    return k


def el_class(it):
    """One Error List item {raw, detail} -> its class: elrule.item_class's local / cond / sr, else input, else
    'other:<el_norm_ocr(raw)>' (the scratch recipes' elcls). Self-test: tools/bench/selftest_t7_ring.py."""
    c = pure("elrule").item_class("{0} {1}".format(it.get("raw", ""), it.get("detail", "")))
    if c:
        return c
    k = el_norm_ocr(it.get("raw"))
    if "unwiredorbadterminal" in k or ("requiredinput" in k and "notwired" in k):
        return "input"
    return "other:" + k


def el_advisory(items, pred_path, plan_path, pred=None):
    """(rc, report) for one Error List read (its items) against the pred's errorlist.per_class (PD354(a), ADVISORY): rc 1 only for
    a class in neither the start read nor the prediction, or |total diff| > EL_TOTAL_TOL; rc 2 when the pred (read from pred_path
    when `pred` is None) is not keyed to plan_path's md5 or has no per_class block; else 0 (the difference is recorded).
    Self-test: el_advisory_selftest / tools/bench/selftest_t7_ring.py."""
    import collections
    pr = pred if pred is not None else jload(pred_path)
    el = pr.get("errorlist") or {}
    pc = el.get("per_class") or {}
    if pred is None and (pr.get("plan") or {}).get("md5") != md5(plan_path):
        return 2, {"why": "pred not keyed to plan md5"}
    if not pc.get("predicted"):
        return 2, {"why": "pred has no errorlist.per_class"}
    start, want = pc["start_read"]["classes"], pc["predicted"]
    got = collections.Counter(el_class(x) for x in items or [])
    diff = dict((c, got.get(c, 0) - want.get(c, 0)) for c in sorted(set(got) | set(want)) if got.get(c, 0) != want.get(c, 0))
    unseen = sorted(c for c in got if c not in start and c not in want)
    td = len(items or []) - int(el["predicted_total"])
    rc = 1 if (unseen or abs(td) > EL_TOTAL_TOL) else 0
    return rc, {"read_total": len(items or []), "pred_total": el["predicted_total"], "total_diff": td, "per_class_diff": diff, "unseen": unseen,
                "verdict": "RETURN" if rc else "RECORDED"}


def el_advisory_print(rc, rep):
    """Print the one EL-ADVISORY line of an el_advisory result."""
    print("EL-ADVISORY total read {0} pred {1} diff {2:+}; per-class diff (read - pred) {3}; unseen classes {4}; verdict {5} rc {6}".format(
        rep.get("read_total"), rep.get("pred_total"), rep.get("total_diff", 0), json.dumps(rep.get("per_class_diff"), sort_keys=True),
        rep.get("unseen"), rep.get("verdict", rep.get("why")), rc), flush=True)


def el_advisory_selftest(pred_path, plan_path):
    """Four synthetic Error List reads built from the pred's predicted per-class counts (offline, the scratch recipes'
    el_selftest): unseen class -> 1; total +4 -> 1; total +2 known -> 0; the prediction itself -> 0. Prints the RESULT line and
    returns 0 / 1."""
    pr = jload(pred_path)
    want = pr["errorlist"]["per_class"]["predicted"]
    txt = {"local": "Local Variable: This variable is not connected to anything.", "sr": "Shift Register: unwired from inside the loop",
           "cond": "While Loop: Conditional terminal is not wired", "input": "Index Array 'Index Array': Contains unwired or bad terminal"}
    mk = lambda c: {"raw": txt[c] if c in txt else c.split(":", 1)[1], "detail": ""}   # noqa: E731
    base = [mk(c) for c, n in sorted(want.items()) for _ in range(n)]
    rt = all(el_class(mk(c)) == c for c in want)
    known = sorted(c for c in want if c.startswith("other:"))[0]
    cases = [("A unseen class (+1 'Feedback Node: Initializer terminal is not wired')", base + [{"raw": "Feedback Node: Initializer terminal is not wired", "detail": ""}], 1),
             ("B total +4 of a known class ({0})".format(known), base + [mk(known)] * 4, 1),
             ("C total +2 of a known class ({0})".format(known), base + [mk(known)] * 2, 0),
             ("D the prediction itself", base, 0)]
    ok = rt
    print("EL-SELFTEST synthetic classes round-trip through elcls: {0} ({1} classes, {2} items)".format(rt, len(want), len(base)), flush=True)
    for name, items, exp in cases:
        rc, rep = el_advisory(items, pred_path, plan_path, pr)
        el_advisory_print(rc, rep)
        print("EL-SELFTEST case {0}: rc {1} expected {2} -> {3}".format(name, rc, exp, "PASS" if rc == exp else "FAIL"), flush=True)
        ok = ok and rc == exp
    print("RESULT " + json.dumps({"schema": "result-line/1", "status": "PASS" if ok else "FAIL", "gates": {"pass": 5 if ok else 0, "fail": 0 if ok else 1},
                                  "first_fail": None if ok else "EL-SELFTEST", "artefacts": []}), flush=True)
    return 0 if ok else 1


def scratch_offline_modes(argv, pred_path, plan_path):
    """The scratch recipes' two OFFLINE modes, before any K.Stage: `--el-advisory-selftest` -> el_advisory_selftest's rc;
    `--el-advisory <read json>` -> EL-ADVISORY of that read's items, its rc. Returns None when neither flag is given."""
    if "--el-advisory-selftest" in argv:
        return el_advisory_selftest(pred_path, plan_path)
    if "--el-advisory" in argv:
        rc, rep = el_advisory(jload(argv[argv.index("--el-advisory") + 1]).get("items") or [], pred_path, plan_path)
        el_advisory_print(rc, rep)
        return rc
    return None


def scratch_tail(st, rc, gone, key, tag, log, base_vi, base_md5, pred_path, plan_path, card, sum_json, fail_mb, launch_mb=680.0):
    """After K.run of a scratch: peak private MB from the log's METER lines vs the pinned X10 (pred_x10), and the gates MEM (peak <=
    fail_mb and <= X10 + 10, PD285(d)/PD328(a)), MEML (peak <= launch_mb, PD331(c)), IN (bed md5 unchanged), GONE (LabVIEW gone);
    writes sum_json. Returns the process rc (rc, else 1 when MEM/IN/GONE fail). The scratch recipes' tail, as calls."""
    import re as _re
    mbs = [float(m) for m in _re.findall(r"METER \w+\s+k\s+\d+\s+private ([\d.]+) MB", open(log, encoding="utf-8", errors="replace").read())] \
        if os.path.exists(log) else []
    peak, X = (max(mbs) if mbs else None), pred_x10(pred_path, plan_path)
    res = st.R.get(key) or {}
    fin = res.get("final")
    print("SCRATCH-{0} peak {1} MB (X10 {2}, d {3}); final {4} md5 {5}; bed md5 {6}".format(
        tag, peak, X, None if peak is None or X is None else round(peak - X, 1), fin, res.get("md5"), md5(base_vi)), flush=True)
    ok_mem = peak is not None and X is not None and peak <= fail_mb and peak <= X + 10.0
    ok_in = md5(base_vi) == base_md5
    print("GATE MEM peak <= {1} and peak <= X10 + 10 (PD285(d), PD328(a)): {0}".format("PASS" if ok_mem else "FAIL", fail_mb), flush=True)
    print("GATE MEML peak <= {1:g} (launch condition, PD331(c)): {0}".format("PASS" if peak is not None and peak <= launch_mb else "FAIL", launch_mb), flush=True)
    print("GATE IN bed md5 unchanged: {0}".format("PASS" if ok_in else "FAIL"), flush=True)
    print("GATE GONE LabVIEW gone: {0}".format("PASS" if gone else "FAIL"), flush=True)
    with open(sum_json, "w", encoding="utf-8") as f:
        json.dump({"card": card, "rc": rc, "peak_mb": peak, "x10": X, "final": fin, "final_md5": res.get("md5"),
                   "inputs_ok": ok_in, "gone": gone, "census_all": st.R.get("census_all")}, f, indent=1)
    return rc or (0 if (ok_mem and ok_in and gone) else 1)
